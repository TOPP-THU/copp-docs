15/// COPP objective descriptor kind shared by second- and third-order solvers.
16#[repr(C)]
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18pub enum CoppObjectiveKind {
19    /// Time objective: `weight * integral dt`.
20    Time = 0,
21    /// Linear objective over solver-specific profile variables.
22    ///
23    /// For COPP2, `alpha` is node-based over `a` and `beta` is interval-based
24    /// over second-order `b`.
25    /// For COPP3, both `alpha` and `beta` are node-based over third-order
26    /// `a` and `b`.
27    Linear = 1,
28    /// Thermal-energy objective.
29    ThermalEnergy = 2,
30    /// Total-variation torque objective.
31    TotalVariationTorque = 3,
32}
33
34/// Borrowed objective descriptor for COPP solvers.
35///
36/// For `Time`, only `weight` is used.
37/// For `Linear`, `alpha` and `beta` are interpreted by each solver family:
38/// COPP2 uses node-based `alpha` and interval-based `beta`, while COPP3 uses
39/// node-based `alpha` and node-based `beta`.
40/// For `ThermalEnergy` and `TotalVariationTorque`, `normalize` must have
41/// length equal to the robot dimension. Torque is evaluated through the robot's
42/// stored inverse-dynamics callback when provided, otherwise by the default
43/// point model (`tau = ddq`).
44#[repr(C)]
45#[derive(Clone, Copy, Debug)]
46pub struct CoppObjective {
47    /// Objective kind.
48    pub kind: CoppObjectiveKind,
49    /// Objective weight.
50    pub weight: f64,
51    /// Linear objective coefficient for variable `a`.
52    pub alpha: CoppSliceF64,
53    /// Linear objective coefficient for variable `b`.
54    pub beta: CoppSliceF64,
55    /// Torque objective normalization vector.
56    pub normalize: CoppSliceF64,
57}
58
59impl CoppObjective {
60    pub(crate) unsafe fn as_rust_objective<'a>(
61        &self,
62        allow_total_variation_torque: bool,
63    ) -> Result<RustCoppObjective<'a>, CoppStatus> {
64        match self.kind {
65            CoppObjectiveKind::Time => Ok(RustCoppObjective::Time(self.weight)),
66            CoppObjectiveKind::Linear => {
67                // SAFETY: The C ABI contract requires non-empty slices to
68                // point to valid contiguous `double` arrays for this call.
69                let alpha = unsafe { self.alpha.as_slice()? };
70                // SAFETY: Same input-slice contract as above.
71                let beta = unsafe { self.beta.as_slice()? };
72                Ok(RustCoppObjective::Linear(self.weight, alpha, beta))
73            }
74            CoppObjectiveKind::ThermalEnergy => {
75                // SAFETY: The C ABI contract requires non-empty slices to
76                // point to valid contiguous `double` arrays for this call.
77                let normalize = unsafe { self.normalize.as_slice()? };
78                Ok(RustCoppObjective::ThermalEnergy(self.weight, normalize))
79            }
80            CoppObjectiveKind::TotalVariationTorque => {
81                if !allow_total_variation_torque {
82                    return Err(CoppStatus::InvalidArgument);
83                }
84                // SAFETY: The C ABI contract requires non-empty slices to
85                // point to valid contiguous `double` arrays for this call.
86                let normalize = unsafe { self.normalize.as_slice()? };
87                Ok(RustCoppObjective::TotalVariationTorque(
88                    self.weight,
89                    normalize,
90                ))
91            }
92        }
93    }
94}

142/// Owned third-order profile returned by TOPP3/COPP3 C ABI solvers.
143///
144/// `a[k]` stores `dot{s}_k^2`, `b[k]` stores `ddot{s}_k`, and
145/// `num_stationary_*` stores the effective stationary head/tail interval
146/// counts returned by the solver.  Both vectors are library-owned and must be
147/// released together with `copp_profile_3rd_free`.
148#[repr(C)]
149#[derive(Debug)]
150pub struct CoppProfile3rd {
151    /// Node-based profile `a[k] = dot{s}_k^2`.
152    pub a: CoppVecF64,
153    /// Node-based profile `b[k] = ddot{s}_k`.
154    pub b: CoppVecF64,
155    /// Effective stationary interval count at the start.
156    pub num_stationary_start: usize,
157    /// Effective stationary interval count at the end.
158    pub num_stationary_end: usize,
159}
160
161impl CoppProfile3rd {
162    pub(crate) const fn empty() -> Self {
163        Self {
164            a: CoppVecF64::empty(),
165            b: CoppVecF64::empty(),
166            num_stationary_start: 0,
167            num_stationary_end: 0,
168        }
169    }
170
171    pub(crate) fn from_parts(a: Vec<f64>, b: Vec<f64>, num_stationary: (usize, usize)) -> Self {
172        Self {
173            a: CoppVecF64::from_vec(a),
174            b: CoppVecF64::from_vec(b),
175            num_stationary_start: num_stationary.0,
176            num_stationary_end: num_stationary.1,
177        }
178    }
179
180    pub(crate) fn free(self) {
181        self.a.free();
182        self.b.free();
183    }
184}

186/// Release a third-order profile returned by COPP.
187///
188/// # Safety
189/// `profile` must either be empty/null or have been returned by a successful
190/// third-order C ABI solver call.  Passing arbitrary pointers or modified
191/// capacity fields is invalid.
192#[unsafe(no_mangle)]
193pub unsafe extern "C" fn copp_profile_3rd_free(profile: CoppProfile3rd) {
194    profile.free();
195    clear_last_error();
196}

198/// Borrowed value descriptor for a TOPP3 problem solved from C.
199///
200/// This is not an owning handle. `robot` and `a_linearization` are borrowed
201/// only during the solver call.  Building the internal TOPP3 problem mutates the
202/// robot's internal linearized third-order constraint cache.
203#[repr(C)]
204#[derive(Clone, Copy, Debug)]
205pub struct Topp3Problem {
206    /// Mutable robot handle that owns station-indexed constraint storage.
207    pub robot: *mut CoppRobot,
208    /// Start station index. The final index is derived from
209    /// `a_linearization.len - 1`.
210    pub idx_s_start: usize,
211    /// Reference profile `a[k]` used to linearize third-order constraints.
212    pub a_linearization: CoppSliceF64,
213    /// Boundary value `a(idx_s_start) = dot{s}^2`.
214    pub a_start: f64,
215    /// Boundary value `a(idx_s_final) = dot{s}^2`.
216    pub a_final: f64,
217    /// Boundary value `b(idx_s_start) = ddot{s}`.
218    pub b_start: f64,
219    /// Boundary value `b(idx_s_final) = ddot{s}`.
220    pub b_final: f64,
221    /// User-input stationary upper bound at the start.
222    pub num_stationary_max_start: usize,
223    /// User-input stationary upper bound at the end.
224    pub num_stationary_max_end: usize,
225    /// Denominator floor used when linearizing third-order constraints.
226    pub a_linearization_floor: f64,
227}
228
229impl Topp3Problem {
230    /// Convert this C descriptor into an internal TOPP3 problem for one solver call.
231    ///
232    /// # Safety
233    /// `self.robot` must be a valid mutable `CoppRobot` handle and
234    /// `self.a_linearization` must be valid for reads during the call.
235    pub(crate) unsafe fn build_rust<'a>(self) -> Result<RustTopp3Problem<'a>, CoppStatus> {
236        // SAFETY: The C ABI contract requires `robot` to be a live mutable
237        // handle for the duration of this call.
238        let robot = unsafe { CoppRobot::robot_mut(self.robot) }.ok_or(CoppStatus::NullPointer)?;
239        // SAFETY: The C ABI contract requires non-empty slices to point to
240        // valid contiguous `double` arrays for the duration of this call.
241        let a_linearization = unsafe { self.a_linearization.as_slice()? };
242
243        RustTopp3ProblemBuilder::new(
244            robot,
245            self.idx_s_start,
246            a_linearization,
247            (self.a_start, self.a_final),
248            (self.b_start, self.b_final),
249        )
250        .with_num_stationary_max_pair((self.num_stationary_max_start, self.num_stationary_max_end))
251        .with_a_linearization_floor(self.a_linearization_floor)
252        .build_with_linearization()
253        .map_err(|error| CoppStatus::from(&error))
254    }
255}

291impl Copp3Problem {
292    /// Convert borrowed C objective descriptors to internal objective descriptors.
293    ///
294    /// # Safety
295    /// `self.objectives` must be valid for `self.num_objectives` reads when
296    /// `self.num_objectives` is non-zero. Any non-empty objective slices must
297    /// be valid for reads during the call.
298    pub(crate) unsafe fn rust_objectives<'a>(
299        self,
300        allow_total_variation_torque: bool,
301    ) -> Result<Vec<RustCoppObjective<'a>>, CoppStatus> {
302        // SAFETY: Objective descriptors are borrowed only during this call.
303        let objectives = unsafe { objective_slice(self.objectives, self.num_objectives)? };
304        objectives
305            .iter()
306            .map(|objective| {
307                // SAFETY: Objective descriptors and their borrowed slices are
308                // used only during this call.
309                unsafe { objective.as_rust_objective(allow_total_variation_torque) }
310            })
311            .collect()
312    }
313
314    /// Build an internal COPP3 problem from the C robot model and run `solve` immediately.
315    ///
316    /// The closure form avoids returning an internal problem that borrows the
317    /// temporary converted objective vector.
318    ///
319    /// # Safety
320    /// Same pointer and slice validity requirements as `Topp3Problem` plus
321    /// the objective requirements documented by the objective conversion helper.
322    pub(crate) unsafe fn with_rust_problem<R>(
323        self,
324        allow_total_variation_torque: bool,
325        solve: impl for<'a> FnOnce(RustCopp3Problem<'a, CoppTorqueForC>) -> Result<R, CoppStatus>,
326    ) -> Result<R, CoppStatus> {
327        // SAFETY: The C ABI contract requires `robot` to be a live mutable
328        // handle for the duration of this call.
329        let robot = unsafe { CoppRobot::robot_mut(self.robot) }.ok_or(CoppStatus::NullPointer)?;
330        // SAFETY: The C ABI contract requires non-empty slices to point to
331        // valid contiguous `double` arrays for the duration of this call.
332        let a_linearization = unsafe { self.a_linearization.as_slice()? };
333        // SAFETY: Objective descriptors are borrowed only during this call.
334        let objectives = unsafe { self.rust_objectives(allow_total_variation_torque)? };
335
336        let problem = RustCopp3ProblemBuilder::new(
337            robot,
338            &objectives,
339            self.idx_s_start,
340            a_linearization,
341            (self.a_start, self.a_final),
342            (self.b_start, self.b_final),
343        )
344        .with_num_stationary_max_pair((self.num_stationary_max_start, self.num_stationary_max_end))
345        .with_a_linearization_floor(self.a_linearization_floor)
346        .build_with_linearization()
347        .map_err(|error| CoppStatus::from(&error))?;
348
349        solve(problem)
350    }
351}

353/// # Safety
354/// When `num_objectives > 0`, `objectives` must point to at least
355/// `num_objectives` initialized `CoppObjective` values that remain valid for
356/// the returned lifetime. The caller must choose a lifetime no longer than the
357/// C-provided storage.
358pub(crate) unsafe fn objective_slice<'a>(
359    objectives: *const CoppObjective,
360    num_objectives: usize,
361) -> Result<&'a [CoppObjective], CoppStatus> {
362    if num_objectives == 0 {
363        return Ok(&[]);
364    }
365    if objectives.is_null() {
366        return Err(CoppStatus::NullPointer);
367    }
368
369    // SAFETY: The caller guarantees that `objectives` is valid for
370    // `num_objectives` reads when `num_objectives` is non-zero.
371    Ok(unsafe { slice::from_raw_parts(objectives, num_objectives) })
372}
