6const COPP_VERSION: &[u8] = concat!(env!("CARGO_PKG_VERSION"), "\0").as_bytes();
7
8/// Return the COPP library version.
9///
10/// The returned pointer is a static null-terminated string owned by the library.
11#[unsafe(no_mangle)]
12pub extern "C" fn copp_version() -> *const c_char {
13    COPP_VERSION.as_ptr().cast()
14}
15
16/// Return a short static message for a COPP status code.
