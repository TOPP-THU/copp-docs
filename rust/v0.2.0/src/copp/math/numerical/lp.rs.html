13/// Safety multiplier used when propagating tolerance across dimension reductions.
14const EPS_SCALE: f64 = 10.0;
15/// Near-zero threshold for feasibility/normalization branch decisions.
16pub(crate) const EPS_ZERO: f64 = 1e-9;
17/// Lower bound for coefficient normalization denominators.
18const EPS_NORMALIZE: f64 = 1e-3;
19/// Default box bound used to stabilize unbounded LP directions.
20pub(crate) const LP_BOUND: f64 = 1e6;

22/// LP tolerance bundle for numerical-robustness controls.
23///
24/// This mirrors the options naming style used in `src/copp` modules
25/// (`*Options` + `*OptionsBuilder`) so tolerance policies are explicit and
26/// configurable.
27#[derive(Clone, Copy, Debug)]
28pub(crate) struct LpToleranceOptions {
29    /// Base feasibility tolerance used by LP inequality checks.
30    pub(crate) feas_tol: f64,
31    /// Lower bound used in row normalization to avoid aggressive scaling.
32    /// Tolerance scale factor used when reducing dimension (3D->2D->1D).
33    pub(crate) reduce_dim_scale: f64,
34}
35
36/// Builder for [`LpToleranceOptions`].
37pub(crate) struct LpToleranceOptionsBuilder {
38    pub feas_tol: f64,
39    pub reduce_dim_scale: f64,
40}

89/// Normalize 2D half-space rows by normal-vector magnitude.
90///
91/// Degenerate rows are mapped to zeros.
92pub(crate) fn normalize_lp2d(a_b: &mut [(f64, f64, f64)]) {
93    for w in a_b.iter_mut() {
94        let norm = (w.0 * w.0 + w.1 * w.1).sqrt();
95        *w = if norm < EPS_ZERO {
96            (0.0, 0.0, 0.0)
97        } else {
98            let norm_inv = 1.0 / norm.max(EPS_NORMALIZE);
99            (w.0 * norm_inv, w.1 * norm_inv, w.2 * norm_inv)
100        };
101    }
102}

104/// Linear programming in 2D plane based on incremental method (only maximize Y).
105/// max Y, s.t. Ab[i].0*X+Ab[i].1*Y<=Ab[i].2
106/// return (X,Y)
107#[inline(always)]
108fn lp_2d_incre_max_y_core<C: Lp2dIncCollector, W: WarmStartLp2d, const NORMALIZE: bool>(
109    a_b: &mut [(f64, f64, f64)],
110    warm_start: &W,
111    tol: &LpToleranceOptions,
112    mut collector: C,
113) -> (f64, f64) {
114    let epsilon = tol.feas_tol;
115    if a_b.len() <= 1 {
116        if a_b.len() == 1 {
117            let a_b_0 = a_b.first().unwrap();
118            if a_b_0.0.abs() < EPS_ZERO && a_b_0.1 > EPS_ZERO {
119                // Unbounded
120                return (f64::NAN, a_b_0.2 / a_b_0.1);
121            }
122        }
123        // Infeasible
124        return (f64::NAN, f64::NAN);
125    }
126
127    if NORMALIZE {
128        normalize_lp2d(a_b);
129    }
130
131    let (mut x, mut y) = warm_start.get_initial_point();
132    let tol_1d = LpToleranceOptions::with_feas_tol(epsilon * tol.reduce_dim_scale);
133    for (i, &(a, b, c)) in warm_start.iter_skip(a_b.iter().enumerate()) {
134        // a*x + b*y <= c
135        if a * x + b * y > c {
136            // Infeasible for this constraint
137            // Let a*x + b*y == c
138            // Apply 1-dim LP
139            if a.abs() < EPS_ZERO && b.abs() < EPS_ZERO {
140                // 0 <= c
141                if c < -epsilon {
142                    collector.clear();
143                    return (f64::NAN, f64::NAN); // Infeasible
144                }
145            } else if a.abs() > b.abs() {
146                // a*x + b*y == c
147                // x == c/a - (b/a)*y == p*y - q
148                let a_inv = 1.0 / a;
149                let p = -b * a_inv;
150                let q = -c * a_inv;
151                // a_*x + b_*y <= c_
152                // a_ * (p*y - q) + b_*y <= c_
153                // (a_*p + b_)*y <= c_ + a_*q
154                let (ymax, _) = lp_1d_core::<_, true>(
155                    a_b.iter()
156                        .take(i)
157                        .map(|&(a_, b_, c_)| (a_ * p + b_, c_ + a_ * q)),
158                    &tol_1d,
159                    &mut collector,
160                );
161                if ymax.is_nan() {
162                    collector.clear();
163                    return (f64::NAN, f64::NAN); // Infeasible
164                }
165                y = ymax.min(y);
166                x = p * y - q;
167                collector.collect_2d_id1(i);
168            } else {
169                // a*x + b*y == c
170                // y == c/b - (a/b)*x == p*x - q
171                let b_inv = 1.0 / b;
172                let p = -a * b_inv;
173                let q = -c * b_inv;
174                // a_*x + b_*y <= c_
175                // a_ * x + b_ * (p*x - q) <= c_
176                // (a_ + b_*p)*x <= c_ + b_*q
177                let (mut xmax, mut xmin) = lp_1d_core::<_, false>(
178                    a_b.iter()
179                        .take(i)
180                        .map(|(a_, b_, c_)| (a_ + b_ * p, c_ + b_ * q)),
181                    &tol_1d,
182                    &mut collector,
183                );
184                if xmax < xmin {
185                    if p.abs() * (xmin - xmax) > epsilon {
186                        collector.clear();
187                        collector.collect_2d_id0(i);
188                        return (f64::NAN, f64::NAN); // Infeasible
189                    } else {
190                        xmax = 0.5 * (xmax + xmin);
191                        xmin = xmax;
192                    }
193                }
194                x = if p > 0.0 {
195                    collector.collect_2d_id1(i);
196                    if xmax.is_finite() {
197                        xmax
198                    } else {
199                        xmin.max(LP_BOUND)
200                    }
201                } else if p < 0.0 {
202                    collector.collect_2d_id0(i);
203                    if xmin.is_finite() {
204                        xmin
205                    } else {
206                        xmax.min(-LP_BOUND)
207                    }
208                } else if xmin > 0.0 {
209                    collector.collect_2d_id0(i);
210                    xmin
211                } else if xmax < 0.0 {
212                    collector.collect_2d_id1(i);
213                    xmax
214                } else {
215                    collector.clear();
216                    collector.collect_2d_id0(i);
217                    0.0
218                };
219                y = p * x - q;
220            }
221        }
222    }
223
224    (x, y)
225}

227/// Linear programming in 2D plane based on incremental method (only maximize Y).
228/// max Y, s.t. Ab[i].0*X+Ab[i].1*Y<=Ab[i].2
229/// return (X,Y)
230pub(crate) fn lp_2d_incre_max_y<W: WarmStartLp2d, const NORMALIZE: bool>(
231    a_b: &mut [(f64, f64, f64)],
232    warm_start: &W,
233    tol: &LpToleranceOptions,
234) -> (f64, f64) {
235    lp_2d_incre_max_y_core::<_, _, NORMALIZE>(a_b, warm_start, tol, SilentLpIncCollector)
236}

238/// Linear programming in 1D line.
239/// a_b[i].0 * x <= a_b[i].1
240/// Return (xmax, xmin) if feasible; otherwise (NaN, NaN).
241#[inline(always)]
242fn lp_1d_core<C: Lp1dIncCollector, const AUTONAN: bool>(
243    a_b: impl Iterator<Item = (f64, f64)>,
244    tol: &LpToleranceOptions,
245    mut collector: C,
246) -> (f64, f64) {
247    let epsilon = tol.feas_tol;
248    // Find the maximum y such that a*x <= b
249    let mut xmax = f64::INFINITY;
250    let mut xmin = -f64::INFINITY;
251
252    for (k, (a, b)) in C::enumerate(a_b) {
253        if a.abs() < b.abs().clamp(EPS_NORMALIZE, 1.0) * EPS_ZERO {
254            // 0 <= b
255            if b < -epsilon {
256                return (f64::NAN, f64::NAN); // Infeasible
257            }
258        } else if a > 0.0 {
259            // x <= x_
260            if b < xmax * a {
261                xmax = b / a;
262                collector.collect_1d_id_max(k.get_key());
263            }
264        } else {
265            // x >= x_
266            if b < xmin * a {
267                xmin = b / a;
268                collector.collect_1d_id_min(k.get_key());
269            }
270        }
271    }
272    if AUTONAN {
273        if xmin <= xmax + epsilon {
274            if xmin <= xmax {
275                (xmax, xmin)
276            } else {
277                let x = 0.5 * (xmax + xmin);
278                (x, x)
279            }
280        } else {
281            (f64::NAN, f64::NAN)
282        }
283    } else {
284        (xmax, xmin)
285    }
286}

288/// Linear programming in 1D line.
289/// a_b[i].0 * x <= a_b[i].1
290/// Return (xmax, xmin) if feasible; otherwise (NaN, NaN).
291#[inline(always)]
292pub(crate) fn lp_1d<const AUTONAN: bool>(
293    a_b: impl Iterator<Item = (f64, f64)>,
294    tol: &LpToleranceOptions,
295) -> (f64, f64) {

297}

299/// Warm-start interface for 2D incremental LP.
300pub(crate) trait WarmStartLp2d {
301    /// Initial feasible guess.
302    fn get_initial_point(&self) -> (f64, f64);
303    /// Iterate constraints while skipping known-prefix constraints if desired.
304    fn iter_skip<I>(&self, a_b: I) -> impl Iterator<Item = I::Item>
305    where
306        I: Iterator;
307}
308/// 2D warm-start policy with explicit initial point and skip length.
309pub(crate) struct Lp2dWarmStart {
310    /// Initial point in transformed 2D LP coordinates.
311    pub x0: (f64, f64),
312    /// Number of leading constraints to skip.
313    pub skip: usize,
314}
315impl WarmStartLp2d for Lp2dWarmStart {
316    #[inline(always)]
317    fn get_initial_point(&self) -> (f64, f64) {
318        self.x0
319    }
320    #[inline(always)]
321    fn iter_skip<I>(&self, a_b: I) -> impl Iterator<Item = I::Item>
322    where
323        I: Iterator,
324    {
325        a_b.skip(self.skip)
326    }
327}

345/// Collector interface for 1D incremental LP diagnostics (active indices).
346trait Lp1dIncCollector {
347    type Key: Lp1dKey;
348    fn collect_1d_id_max(&mut self, index: usize);
349    fn collect_1d_id_min(&mut self, index: usize);
350    fn clear(&mut self);
351    fn enumerate<D, I>(a_b: I) -> impl Iterator<Item = (Self::Key, D)>
352    where
353        I: Iterator<Item = D>;
354}
355impl<T: Lp1dIncCollector> Lp1dIncCollector for &mut T {
356    type Key = T::Key;
357    #[inline(always)]
358    fn collect_1d_id_max(&mut self, k: usize) {
359        (**self).collect_1d_id_max(k);
360    }
361    #[inline(always)]
362    fn collect_1d_id_min(&mut self, k: usize) {
363        (**self).collect_1d_id_min(k);
364    }
365    #[inline(always)]
366    fn clear(&mut self) {
367        (**self).clear();
368    }
369    #[inline(always)]
370    fn enumerate<D, I>(a_b: I) -> impl Iterator<Item = (Self::Key, D)>
371    where
372        I: Iterator<Item = D>,
373    {
374        T::enumerate(a_b)
375    }
376}

417/// Solve 2*2 linear equations: a*x=b.
418pub(crate) fn solve_2x2(a: ((f64, f64), (f64, f64)), b: (f64, f64)) -> Option<(f64, f64)> {
419    let det = cross_product_2d((a.0.0, a.0.1), (a.1.0, a.1.1));
420    if det.abs() < f64::EPSILON {
421        return None; // Singular matrix
422    }
423    let det_inv = 1.0 / det;
424    let x1 = cross_product_2d((b.0, a.0.1), (b.1, a.1.1)) * det_inv;
425    let x2 = cross_product_2d((a.0.0, b.0), (a.1.0, b.1)) * det_inv;
426    Some((x1, x2))
427}
