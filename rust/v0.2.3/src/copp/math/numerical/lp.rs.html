3//! # Method identity
4//! The functions in this module are low-level numeric engines used by TOPP/COPP
5//! planners. They operate on half-space form constraints and provide:
6//! - direct 1D/2D incremental LP solves,
7//! - warm-start interfaces,
8//! - helper normalization and tiny linear-system primitives.

13/// Safety multiplier used when propagating tolerance across dimension reductions.
14const EPS_SCALE: f64 = 10.0;
15/// Near-zero threshold for feasibility/normalization branch decisions.
16pub(crate) const EPS_ZERO: f64 = 1e-9;
17/// Lower bound for coefficient normalization denominators.
18const EPS_NORMALIZE: f64 = 1e-3;
19/// Initial half-width of the box the incremental LP kernels pose their problems
20/// over, and the sentinel value of a warm start that means "start beyond the
21/// set in the objective direction". The box grows by [`LP_BOX_GROWTH`] whenever
22/// an answer lands on its edge, so this is a starting point, not a limit.
23pub(crate) const LP_BOUND: f64 = 1e6;
24/// Relative floating-point tolerance (about 450 ulps) for cancellation-sensitive
25/// comparisons: a quantity computed as a difference of terms is treated as zero
26/// when it is below `EPS_ROUNDING` times the magnitude of those terms.
27///
28/// This is deliberately not the feasibility tolerance. A violation at the
29/// rounding level of `a*x + b*y - c` carries no information about the row, and
30/// pivoting onto it breaks the incremental invariant when the row is nearly
31/// parallel to the active one; a violation above it is real and must pivot,
32/// however small it is in absolute terms.
33const EPS_ROUNDING: f64 = 1e-13;
34
35/// Growth factor of the implicit box when an answer lands on its edge.
36///
37/// `LP_BOUND` is not a guard against unbounded problems, it is part of the
38/// problem statement, and the incremental solvers start from its edge and can
39/// only move inwards. A problem whose optimum lies beyond it therefore gets the
40/// box edge as its answer: with the path parameter in millimetres `a = sdot^2`
41/// passes `1e6` at 1000 mm/s, and the profile was observed to stick there. An
42/// answer on the box is the signature of that, so the box grows and the solve
43/// repeats; only an unbounded problem keeps growing to the cap.
44const LP_BOX_GROWTH: f64 = 1e3;
45/// How many times the box may grow before the answer is taken as it is; an
46/// unbounded problem is then reported at the edge of a `1e24` box.
47const LP_BOX_GROWTH_STEPS: usize = 6;
48/// Fraction of the box half-width beyond which an answer counts as sitting on
49/// the box. Once the box has grown past the problem, a genuine optimum lies
50/// well inside it (`LP_BOX_GROWTH` is `1e3`, the threshold one half).
51const LP_BOX_SATURATION: f64 = 0.5;
52
53/// Whether a two-dimensional answer sits on the box: a bounded problem whose
54/// answer does was clipped by it, and an unbounded one has nowhere else to go.
55#[inline(always)]
56fn lp_2d_saturated(x: f64, y: f64, bound: f64) -> bool {
57    let edge = LP_BOX_SATURATION * bound;
58    x.is_finite() && y.is_finite() && (y >= edge || x.abs() >= edge)
59}

61/// Error bound for the sign of a 2x2 determinant, as a multiple of the sum of
62/// the magnitudes of the two products that cancel into it.
63///
64/// This is Shewchuk's `ccwerrboundA = (3 + 16*eps)*eps` with `eps = 2^-53`, the
65/// bound under which the floating-point sign of `orient2d` is proven reliable
66/// (Shewchuk, *Adaptive Precision Floating-Point Arithmetic and Fast Robust
67/// Geometric Predicates*, DCG 18(3), 1997). Comparing the determinant against a
68/// fixed epsilon instead would make the test depend on how the rows happen to
69/// be scaled: a determinant of `1e-12` is decisive for rows of magnitude
70/// `1e-8` and pure noise for rows of magnitude `1e3`.
71const CCW_ERRBOUND_A: f64 = 3.330_669_073_875_47e-16;

139/// Scale each 2D half-space row to a unit normal, so that its right-hand side
140/// becomes the signed distance from the origin to its boundary.
141///
142/// Rows with no normal at all are mapped to zeros; they carry no direction and
143/// no point can violate them.
144///
145/// The divisor is the row's own norm and nothing else. Clamping it from below —
146/// as this did, at `EPS_NORMALIZE` — leaves a row of norm `1e-10` at norm
147/// `1e-7` rather than at 1, so every absolute tolerance downstream is inflated
148/// by `1e-3 / norm` on that row, up to a factor of a million. The consequence
149/// is not a loss of accuracy but a loss of the constraint: multiplying one row
150/// by `1e-10` changes nothing geometrically, yet was observed to move the
151/// objective from `-6.91e2` to `+4.48e4` because the row stopped being seen.
152pub(crate) fn normalize_lp2d(a_b: &mut [(f64, f64, f64)]) {
153    for w in a_b.iter_mut() {
154        let norm = (w.0 * w.0 + w.1 * w.1).sqrt();
155        *w = if norm > 0.0 && norm.is_finite() {
156            let norm_inv = 1.0 / norm;
157            (w.0 * norm_inv, w.1 * norm_inv, w.2 * norm_inv)
158        } else {
159            (0.0, 0.0, 0.0)
160        };
161    }
162}

164/// Relative tolerance for calling a cancelled coefficient zero.
165///
166/// The coefficient reaching [`lp_1d_core`] from [`incre_step_2d`] is a
167/// difference of two products, so this is the same question `solve_2x2` asks of
168/// its determinant and takes the same bound.
169const LP1D_CANCEL_EPS: f64 = CCW_ERRBOUND_A;

171/// How many times a row found violated after the incremental pass may be
172/// reprocessed. Exact arithmetic needs one; the rest is room for rounding to
173/// bounce a pair of rows against each other.
174const MAX_DEFERRED_ROUNDS: usize = 4;

176/// Target a smaller residual than the public feasibility tolerance when the
177/// rare degeneracy recovery path has already paid for a full prefix scan.  The
178/// remaining margin lets later reductions accumulate some rounding without
179/// immediately turning the repaired row into another violation.
180const LP2D_REPAIR_TOL_SCALE: f64 = 0.25;

182/// One incremental step: pin row `row` as an equality and optimize along it
183/// subject to rows `0..upto`, moving `(x, y)` onto the result.
184///
185/// Returns `false` when that reveals the system to be infeasible.
186///
187/// `upto` is separate from `row` so the same step serves both phases: during
188/// the forward pass a row is optimized against the rows already seen, and
189/// during the deferred correction against all of them.
190#[inline(always)]
191#[allow(clippy::too_many_arguments)]
192fn incre_step_2d<C: Lp2dIncCollector>(
193    a_b: &[(f64, f64, f64)],
194    row: usize,
195    upto: usize,
196    x: &mut f64,
197    y: &mut f64,
198    epsilon: f64,
199    tol_1d: &LpToleranceOptions,
200    collector: &mut C,
201    bound: f64,
202) -> bool {
203    let (a, b, c) = a_b[row];
204    let i = row;
205    if a.abs() < EPS_ZERO && b.abs() < EPS_ZERO {
206        // 0 <= c
207        return c >= -epsilon;
208    }
209    if a.abs() > b.abs() {
210        // a*x + b*y == c
211        // x == c/a - (b/a)*y == p*y - q
212        let a_inv = 1.0 / a;
213        let p = -b * a_inv;
214        let q = -c * a_inv;
215        // a_*x + b_*y <= c_
216        // a_ * (p*y - q) + b_*y <= c_
217        // (a_*p + b_)*y <= c_ + a_*q
218        let (ymax, _) = lp_1d_core::<_, true, true>(
219            a_b.iter().take(upto).map(|&(a_, b_, c_)| {
220                let term = a_ * p;
221                (term + b_, c_ + a_ * q, term.abs() + b_.abs())
222            }),
223            tol_1d,
224            &mut *collector,
225        );
226        if ymax.is_nan() {
227            return false;
228        }
229        *y = ymax.min(*y);
230        *x = p * *y - q;
231        collector.collect_2d_id1(i);
232        return true;
233    }
234    // a*x + b*y == c
235    // y == c/b - (a/b)*x == p*x - q
236    let b_inv = 1.0 / b;
237    let p = -a * b_inv;
238    let q = -c * b_inv;
239    // a_*x + b_*y <= c_
240    // a_ * x + b_ * (p*x - q) <= c_
241    // (a_ + b_*p)*x <= c_ + b_*q
242    let (mut xmax, mut xmin) = lp_1d_core::<_, false, true>(
243        a_b.iter().take(upto).map(|(a_, b_, c_)| {
244            let term = b_ * p;
245            (a_ + term, c_ + b_ * q, a_.abs() + term.abs())
246        }),
247        tol_1d,
248        &mut *collector,
249    );
250    if xmax < xmin {
251        // `xmax`/`xmin` bound the same variable `x`, so the inconsistency
252        // `xmin - xmax` lives in `x` units and the tolerance must be applied there.
253        // This branch has `|p| <= 1` (it is the `|a| <= |b|` case), so the previous
254        // `p.abs() * (xmin - xmax)` test measured the inconsistency in `y` units and
255        // was looser by the unbounded factor `1 / |p|`.
256        if xmin - xmax > epsilon.max(EPS_ROUNDING * (xmin.abs() + xmax.abs())) {
257            collector.clear();
258            collector.collect_2d_id0(i);
259            return false;
260        }
261        xmax = 0.5 * (xmax + xmin);
262        xmin = xmax;
263    }
264    *x = if p > 0.0 {
265        collector.collect_2d_id1(i);
266        if xmax.is_finite() {
267            xmax
268        } else {
269            xmin.max(bound)
270        }
271    } else if p < 0.0 {
272        collector.collect_2d_id0(i);
273        if xmin.is_finite() {
274            xmin
275        } else {
276            xmax.min(-bound)
277        }
278    } else if xmin > 0.0 {
279        collector.collect_2d_id0(i);
280        xmin
281    } else if xmax < 0.0 {
282        collector.collect_2d_id1(i);
283        xmax
284    } else {
285        collector.clear();
286        collector.collect_2d_id0(i);
287        0.0
288    };
289    *y = p * *x - q;
290    true
291}

293/// Project only the non-objective coordinate while keeping the current objective
294/// value.  The returned point satisfies `rows[..upto]` within `epsilon`.
295///
296/// In exact arithmetic, if `(x, y)` is optimal for the old prefix, any feasible
297/// `(x_new, y)` is also optimal after adding rows: the feasible set only shrank,
298/// while its old upper bound `y` is still attained.  Here the same argument is
299/// used with the solver's advertised residual tolerance, so the projection adds
300/// no objective loss and restores the corresponding numerical invariant.
301#[cold]
302#[inline(never)]
303fn project_x_at_fixed_y(
304    a_b: &[(f64, f64, f64)],
305    upto: usize,
306    x: f64,
307    y: f64,
308    epsilon: f64,
309) -> Option<f64> {
310    // The common degeneracy needs no coordinate change at all.  Check this
311    // before constructing the interval so a rounded `c + epsilon` cannot lose
312    // a tolerance that the solver's own residual test can still see.
313    if x.is_finite()
314        && a_b
315            .iter()
316            .take(upto)
317            .all(|&(a, b, c)| a * x + b * y - c <= epsilon)
318    {
319        return Some(x);
320    }
321
322    let mut xmin = f64::NEG_INFINITY;
323    let mut xmax = f64::INFINITY;
324
325    for &(a, b, c) in a_b.iter().take(upto) {
326        if !a.is_finite() || !b.is_finite() || !c.is_finite() {
327            return None;
328        }
329        // At fixed y, a*x + b*y <= c + epsilon is a one-dimensional
330        // interval. `mul_add` is deliberately confined to this rare fallback.
331        let rhs = (-b).mul_add(y, c + epsilon);
332        if !rhs.is_finite() {
333            return None;
334        }
335        if a > 0.0 {
336            xmax = xmax.min(rhs / a);
337        } else if a < 0.0 {
338            xmin = xmin.max(rhs / a);
339        } else if rhs < 0.0 {
340            return None;
341        }
342    }
343    if xmin > xmax {
344        return None;
345    }
346
347    let is_prefix_feasible = |x_test: f64| {
348        x_test.is_finite()
349            && a_b
350                .iter()
351                .take(upto)
352                .all(|&(a, b, c)| a * x_test + b * y - c <= epsilon)
353    };
354
355    // Usually the old x is already in the interval (the motivating case is a
356    // one-ulp disagreement between equivalent evaluation orders).  Trying the
357    // projection and then interior/end-point alternatives also covers a bound
358    // rounded by one operation in the opposite direction from the final check.
359    let projected = x.max(xmin).min(xmax);
360    if is_prefix_feasible(projected) {
361        return Some(projected);
362    }
363    if xmin.is_finite() && xmax.is_finite() {
364        let midpoint = 0.5 * xmin + 0.5 * xmax;
365        if is_prefix_feasible(midpoint) {
366            return Some(midpoint);
367        }
368    }
369    if is_prefix_feasible(xmin) {
370        return Some(xmin);
371    }
372    if is_prefix_feasible(xmax) {
373        return Some(xmax);
374    }
375    None
376}

378/// Check that a trial collector describes active rows whose normal cone
379/// supports the `+y` objective.  A lone row is also valid when its positive
380/// y-normal combines with the solver's implicit x box; `y == LP_BOUND` is the
381/// corresponding implicit objective-bound case.
382#[cold]
383#[inline(never)]
384#[allow(clippy::too_many_arguments)]
385fn active_basis_supports_max_y(
386    a_b: &[(f64, f64, f64)],
387    upto: usize,
388    selected_row: usize,
389    x: f64,
390    y: f64,
391    epsilon: f64,
392    ids: (usize, usize),
393    bound: f64,
394) -> bool {
395    let valid0 = ids.0 < upto;
396    let valid1 = ids.1 < upto;
397    if (!valid0 && !valid1)
398        || (selected_row != ids.0 && selected_row != ids.1)
399        || (valid0 && (a_b[ids.0].0 * x + a_b[ids.0].1 * y - a_b[ids.0].2).abs() > epsilon)
400        || (valid1 && (a_b[ids.1].0 * x + a_b[ids.1].1 * y - a_b[ids.1].2).abs() > epsilon)
401    {
402        return false;
403    }
404
405    if !valid0 || !valid1 || ids.0 == ids.1 {
406        let id = if valid0 { ids.0 } else { ids.1 };
407        let (a, b, _) = a_b[id];
408        return y == bound
409            || (b > 0.0 && (a == 0.0 || (a > 0.0 && x == -bound) || (a < 0.0 && x == bound)));
410    }
411
412    let (n0, n1) = (a_b[ids.0], a_b[ids.1]);
413    let (p, q) = (n0.0 * n1.1, n0.1 * n1.0);
414    let det = p - q;
415    if !det.is_finite() || det.abs() <= CCW_ERRBOUND_A * (p.abs() + q.abs()) {
416        return false;
417    }
418    // [n0 n1] * lambda = (0, 1).  Nonnegative multipliers are the
419    // two-dimensional KKT/normal-cone certificate for maximizing y.
420    let (lambda0, lambda1) = (-n1.0 / det, n0.0 / det);
421    lambda0.is_finite() && lambda1.is_finite() && lambda0 >= 0.0 && lambda1 >= 0.0
422}

424/// Deterministically search a prefix by trying every possible active boundary.
425/// This is quadratic in the slow path, but each returned candidate has passed
426/// the full residual check and an active-basis certificate.  Numerical failure
427/// of every candidate is still reported to the caller rather than described as
428/// a proof that the mathematical LP is infeasible.
429#[cold]
430#[inline(never)]
431#[allow(clippy::too_many_arguments)]
432fn rebuild_2d_prefix(
433    a_b: &[(f64, f64, f64)],
434    upto: usize,
435    x_upper: f64,
436    y_upper: f64,
437    y_floor: f64,
438    epsilon: f64,
439    tol_1d: &LpToleranceOptions,
440    bound: f64,
441) -> Option<(f64, f64, (usize, usize))> {
442    let mut best: Option<(f64, f64, (usize, usize))> = None;
443    for row in 0..upto {
444        let (a_row, b_row, c_row) = a_b[row];
445        if a_row.abs() < EPS_ZERO && b_row.abs() < EPS_ZERO {
446            // This row has no boundary that could support an optimum.
447            continue;
448        }
449        let (mut x_trial, mut y_trial) = (x_upper, y_upper);
450        let mut trial_collector = IndexLpIncCollector::new();
451        if !incre_step_2d(
452            a_b,
453            row,
454            upto,
455            &mut x_trial,
456            &mut y_trial,
457            epsilon,
458            tol_1d,
459            &mut trial_collector,
460            bound,
461        ) || !x_trial.is_finite()
462            || !y_trial.is_finite()
463            || y_trial < y_floor
464            || y_trial > y_upper
465            || (a_row * x_trial + b_row * y_trial - c_row).abs() > epsilon
466            || a_b
467                .iter()
468                .take(upto)
469                .any(|&(a, b, c)| a * x_trial + b * y_trial - c > epsilon)
470        {
471            continue;
472        }
473        // Downstream indexed callers require the first id to be the real row
474        // in a one-row basis; preserve both positions for a two-row basis.
475        let ids = if trial_collector.id.0 >= upto && trial_collector.id.1 < upto {
476            (trial_collector.id.1, trial_collector.id.0)
477        } else {
478            (trial_collector.id.0, trial_collector.id.1)
479        };
480        if !active_basis_supports_max_y(a_b, upto, row, x_trial, y_trial, epsilon, ids, bound) {
481            continue;
482        }
483        // `y_upper` came from the optimum of the previous prefix.  Adding a
484        // row cannot improve it, so attaining the same value proves that no
485        // later candidate can win and avoids the quadratic tail in the common
486        // redundant-row recovery.
487        if y_trial == y_upper {
488            return Some((x_trial, y_trial, ids));
489        }
490        if best.is_none_or(|(_, y_best, _)| y_trial > y_best) {
491            best = Some((x_trial, y_trial, ids));
492        }
493    }
494    best
495}

497/// Recover a tolerance-verified optimum/basis candidate for `a_b[..upto]` after
498/// the usual pivot failed.  An arbitrary feasible projection is never returned:
499/// either the old objective level remains attainable, or the active-boundary
500/// scan supplies a verified basis that can be committed to the real collector.
501#[cold]
502#[inline(never)]
503#[allow(clippy::too_many_arguments)]
504fn recover_2d_prefix<C: Lp2dIncCollector>(
505    a_b: &[(f64, f64, f64)],
506    upto: usize,
507    x: f64,
508    y: f64,
509    epsilon: f64,
510    tol_1d: &LpToleranceOptions,
511    collector: &mut C,
512    bound: f64,
513) -> Option<(f64, f64)> {
514    let fixed_y_repair = project_x_at_fixed_y(a_b, upto, x, y, epsilon);
515    let x_seed = fixed_y_repair.unwrap_or(x);
516    let y_floor = if fixed_y_repair.is_some() {
517        // The old y is still both an upper bound and an attainable lower
518        // bound. Permit only the recovery tolerance of reconstruction drift.
519        y - epsilon
520    } else {
521        f64::NEG_INFINITY
522    };
523
524    if let Some((x_rebuilt, y_rebuilt, ids)) =
525        rebuild_2d_prefix(a_b, upto, x_seed, y, y_floor, epsilon, tol_1d, bound)
526    {
527        // All numeric work above used a private collector. Commit the verified
528        // basis only after a winner exists, so failed recovery is transactional
529        // with respect to the caller's active-set state.
530        collector.clear();
531        if ids.0 < upto {
532            collector.collect_2d_id0(ids.0);
533        }
534        if ids.1 < upto {
535            collector.collect_2d_id1(ids.1);
536        }
537        return Some((x_rebuilt, y_rebuilt));
538    }
539
540    // If no alternate boundary survived numerically but the unchanged point
541    // itself passed the complete prefix check, retaining it is still a valid
542    // tolerance-level certificate and leaves the old collector untouched.
543    if fixed_y_repair == Some(x) {
544        return Some((x, y));
545    }
546    None
547}

549/// Linear programming in 2D plane based on incremental method (only maximize Y).
550/// max Y, s.t. Ab[i].0*X+Ab[i].1*Y<=Ab[i].2
551/// return (X,Y)
552///
553/// `bound` is the half-width of the box the problem is posed over, in the
554/// caller's coordinates. An answer on the box is a clipped one when the
555/// problem is bounded, so the box grows and the solve repeats.
556#[inline(always)]
557fn lp_2d_incre_max_y_core<C: Lp2dIncCollector, W: WarmStartLp2d, const NORMALIZE: bool>(
558    a_b: &mut [(f64, f64, f64)],
559    warm_start: &W,
560    tol: &LpToleranceOptions,
561    mut collector: C,
562    mut bound: f64,
563) -> (f64, f64) {
564    let mut saturated = false;
565    for step in 0..=LP_BOX_GROWTH_STEPS {
566        collector.clear();
567        let (x, y) = lp_2d_incre_max_y_once::<_, _, NORMALIZE>(
568            a_b,
569            warm_start,
570            tol,
571            &mut collector,
572            bound,
573            &mut saturated,
574        );
575        if !saturated || step == LP_BOX_GROWTH_STEPS {
576            return (x, y);
577        }
578        bound *= LP_BOX_GROWTH;
579    }
580    unreachable!()
581}

583/// One solve of [`lp_2d_incre_max_y_core`] over a fixed box; `saturated`
584/// reports whether the answer sits on that box.
585#[inline(always)]
586fn lp_2d_incre_max_y_once<C: Lp2dIncCollector, W: WarmStartLp2d, const NORMALIZE: bool>(
587    a_b: &mut [(f64, f64, f64)],
588    warm_start: &W,
589    tol: &LpToleranceOptions,
590    mut collector: C,
591    mut bound: f64,
592    saturated: &mut bool,
593) -> (f64, f64) {
594    *saturated = false;
595    let epsilon = tol.feas_tol;
596    if a_b.len() <= 1 {
597        if a_b.len() == 1 {
598            let a_b_0 = a_b.first().unwrap();
599            if a_b_0.0.abs() < EPS_ZERO && a_b_0.1 > EPS_ZERO {
600                // Unbounded
601                return (f64::NAN, a_b_0.2 / a_b_0.1);
602            }
603        }
604        // Infeasible
605        return (f64::NAN, f64::NAN);
606    }
607
608    if NORMALIZE {
609        normalize_lp2d(a_b);
610    }
611
612    let (mut x, mut y) = warm_start.get_initial_point();
613    if x.is_finite() && y.is_finite() && (x.abs() > bound || y.abs() > bound) {
614        // A warm start is the optimum of its prefix over the box that solve
615        // was posed over; this solve must not pose a smaller one.
616        bound = LP_BOX_GROWTH * x.abs().max(y.abs());
617    }
618    if y == LP_BOUND && bound > LP_BOUND {
619        // The sentinel "start beyond the set" has to sit on the box edge that
620        // is actually in use, or it is below the optimum and the solver, which
621        // only ever moves down, returns the sentinel itself.
622        y = bound;
623    }
624    let tol_1d = LpToleranceOptions::with_feas_tol(epsilon * tol.reduce_dim_scale);
625    // Reported rather than returned from inside the loop: the loop holds `a_b`
626    // immutably, and restoring the rows needs it back.
627    let mut feasible = true;
628    for (i, &(a, b, c)) in warm_start.iter_skip(a_b.iter().enumerate()) {
629        // a*x + b*y <= c, up to the rounding level of the terms.
630        let ax = a * x;
631        let by = b * y;
632        let residual = ax + by - c;
633        if residual > EPS_ROUNDING * (ax.abs() + by.abs() + c.abs()) {
634            // Keep the strict incremental pivot for every genuine violation:
635            // even a small normal residual can move the objective appreciably
636            // when this row is nearly parallel to the objective.  The tolerance
637            // is used only to adjudicate a failed reduction.  In that case the
638            // current point already satisfies the new row to the solver's
639            // advertised accuracy, whereas declaring the whole prefix
640            // infeasible would amplify roundoff between almost coincident rows.
641            //
642            // Trial with a silent collector so a rejected step is transactional;
643            // `lp_1d_core` may update active indices before discovering that its
644            // interval is empty.
645            if residual <= epsilon {
646                let (mut x_trial, mut y_trial) = (x, y);
647                if !incre_step_2d(
648                    a_b,
649                    i,
650                    i,
651                    &mut x_trial,
652                    &mut y_trial,
653                    epsilon,
654                    &tol_1d,
655                    &mut SilentLpIncCollector,
656                    bound,
657                ) {
658                    let upto = i + 1;
659                    let repair_epsilon = epsilon * LP2D_REPAIR_TOL_SCALE;
660                    let repair_tol_1d =
661                        LpToleranceOptions::with_feas_tol(repair_epsilon * tol.reduce_dim_scale);
662                    if let Some(recovered) = recover_2d_prefix(
663                        a_b,
664                        upto,
665                        x,
666                        y,
667                        repair_epsilon,
668                        &repair_tol_1d,
669                        &mut collector,
670                        bound,
671                    ) {
672                        (x, y) = recovered;
673                        continue;
674                    }
675
676                    // A narrow feasible slice need not have enough floating-
677                    // point room for the smaller repair tolerance.  Retrying
678                    // with the public tolerance preserves the solver's existing
679                    // contract without charging the normal path for recovery.
680                    if repair_epsilon != epsilon
681                        && let Some(recovered) = recover_2d_prefix(
682                            a_b,
683                            upto,
684                            x,
685                            y,
686                            epsilon,
687                            &tol_1d,
688                            &mut collector,
689                            bound,
690                        )
691                    {
692                        (x, y) = recovered;
693                        continue;
694                    }
695
696                    feasible = false;
697                    break;
698                }
699            }
700            if !incre_step_2d(
701                a_b,
702                i,
703                i,
704                &mut x,
705                &mut y,
706                epsilon,
707                &tol_1d,
708                &mut collector,
709                bound,
710            ) {
711                feasible = false;
712                break;
713            }
714        }
715    }
716    if !feasible {
717        collector.clear();
718        return (f64::NAN, f64::NAN); // Infeasible
719    }
720
721    // Deferred correction. The pass above is optimal for the rows in the order
722    // it happened to see them, and a row it processed early can be left
723    // violated by a later one when the arithmetic that combined them lost
724    // digits. Rather than tightening a tolerance to prevent that, look at the
725    // answer and ask which row it violates: that is a comparison between two
726    // quantities of the same size, needing no tolerance to interpret, and it is
727    // only available once there is an answer to look at.
728    //
729    // Reprocessing such a row is exact rather than a repair. Seidel's step
730    // pins one row as an equality and optimizes along it subject to the rest,
731    // which yields the optimum of the whole system whenever the optimum lies on
732    // that row -- and it does, precisely because the row is violated at the
733    // current point. Taking `a_b.len()` as the horizon rather than the row's
734    // own index is what makes "the rest" mean every row.
735    //
736    // One round suffices in exact arithmetic; the cap is there because rounding
737    // can send two rows back and forth.
738    for _ in 0..MAX_DEFERRED_ROUNDS {
739        let mut worst = 0.0;
740        let mut worst_row = usize::MAX;
741        for (i, &(a, b, c)) in a_b.iter().enumerate() {
742            let ax = a * x;
743            let by = b * y;
744            let excess = ax + by - c - epsilon.max(EPS_ROUNDING * (ax.abs() + by.abs() + c.abs()));
745            if excess > worst {
746                worst = excess;
747                worst_row = i;
748            }
749        }
750        if worst_row == usize::MAX {
751            break;
752        }
753        crate::verbosity_log!(
754            crate::diag::Verbosity::Debug,
755            "lp_2d deferred correction fired: row {worst_row} violated by {worst:.3e} beyond tolerance"
756        );
757        if !incre_step_2d(
758            a_b,
759            worst_row,
760            a_b.len(),
761            &mut x,
762            &mut y,
763            epsilon,
764            &tol_1d,
765            &mut collector,
766            bound,
767        ) {
768            collector.clear();
769            return (f64::NAN, f64::NAN); // Infeasible
770        }
771    }
772
773    // y = if (y / BOUND - 1.0).abs() < epsilon {
774    //     f64::INFINITY
775    // } else {
776    //     y
777    // };
778    *saturated = lp_2d_saturated(x, y, bound);
779    (x, y)
780}

793/// Linear programming in 1D line.
794/// a_b[i].0 * x <= a_b[i].1
795/// Return (xmax, xmin) if feasible; otherwise (NaN, NaN).
796#[inline(always)]
797fn lp_1d_core<C: Lp1dIncCollector, const AUTONAN: bool, const CANCEL: bool>(
798    a_b: impl Iterator<Item = (f64, f64, f64)>,
799    tol: &LpToleranceOptions,
800    mut collector: C,
801) -> (f64, f64) {
802    let epsilon = tol.feas_tol;
803    // Find the maximum y such that a*x <= b
804    let mut xmax = f64::INFINITY;
805    let mut xmin = -f64::INFINITY;
806
807    for (k, (a, b, cancel_scale)) in C::enumerate(a_b) {
808        // println!("\t\t(LP-1D) a={}, b={}", a, b);
809        let coefficient_is_zero = if CANCEL {
810            a.abs() <= LP1D_CANCEL_EPS * cancel_scale
811        } else {
812            a.abs() < b.abs().clamp(EPS_NORMALIZE, 1.0) * EPS_ZERO
813        };
814        if coefficient_is_zero {
815            // 0 <= b
816            if b < -epsilon {
817                return (f64::NAN, f64::NAN); // Infeasible
818            }
819        } else if a > 0.0 {
820            // x <= x_
821            if b < xmax * a {
822                xmax = b / a;
823                collector.collect_1d_id_max(k.get_key());
824            }
825        } else {
826            // x >= x_
827            if b < xmin * a {
828                xmin = b / a;
829                collector.collect_1d_id_min(k.get_key());
830            }
831        }
832
833        // println!("\t\t(LP-1D) xmax={}, xmin={}, a={}, b={}", xmax, xmin, a, b);
834    }
835    // println!("\t\t(LP-1D) xmax={}, xmin={}", xmax, xmin);
836    if AUTONAN {
837        if xmin <= xmax + epsilon.max(EPS_ROUNDING * (xmin.abs() + xmax.abs())) {
838            if xmin <= xmax {
839                (xmax, xmin)
840            } else {
841                let x = 0.5 * (xmax + xmin);
842                (x, x)
843            }
844        } else {
845            // println!("\t(LP-1D) xmax={}, xmin={}", xmax, xmin);
846            (f64::NAN, f64::NAN)
847        }
848    } else {
849        (xmax, xmin)
850    }
851}

853/// Linear programming in 1D line.
854/// a_b[i].0 * x <= a_b[i].1
855/// Return (xmax, xmin) if feasible; otherwise (NaN, NaN).
856#[inline(always)]
857pub(crate) fn lp_1d<const AUTONAN: bool>(
858    a_b: impl Iterator<Item = (f64, f64)>,
859    tol: &LpToleranceOptions,
860) -> (f64, f64) {

862}

1029/// Solve 2*2 linear equations: a*x=b.
1030pub(crate) fn solve_2x2(a: ((f64, f64), (f64, f64)), b: (f64, f64)) -> Option<(f64, f64)> {
1031    let (p, q) = (a.0.0 * a.1.1, a.0.1 * a.1.0);
1032    let det = p - q;
1033    // The sign of `det` is only meaningful while it stands above the rounding
1034    // of the two products that cancel into it; see [`CCW_ERRBOUND_A`]. The
1035    // non-finite test is what makes this NaN-safe, since every comparison
1036    // against NaN is false.
1037    if !det.is_finite() || det.abs() <= CCW_ERRBOUND_A * (p.abs() + q.abs()) {
1038        return None; // Singular matrix, or non-finite coefficients
1039    }
1040    let det_inv = 1.0 / det;
1041    let x1 = cross_product_2d((b.0, a.0.1), (b.1, a.1.1)) * det_inv;
1042    let x2 = cross_product_2d((a.0.0, b.0), (a.1.0, b.1)) * det_inv;
1043    Some((x1, x2))
1044}

1152            (6.23700623703079060e1, -6.23700623703079060e1, 3.0e2),
1153            (0.0, 0.0, 3.0e2),
1154            (0.0, 0.0, 3.0e2),
1155            (-6.65280665283189023e1, 6.65280665283248140e1, 3.0e2),
1156            (6.65280665283189023e1, -6.65280665283248140e1, 3.0e2),
1157        ]
1158        .into();
1159
1160        // `backward_bound_a_next` normalizes the rows, then solves with
1161        // `NORMALIZE = false`; `reach_set2` runs it with `lp_feas_tol = 1e-8` and
1162        // starts from the midpoint of the successor interval.
1163        let tol = LpToleranceOptions::with_feas_tol(1.0e-8);
1164        normalize_lp2d(&mut a_b);
1165        let warm_start = Lp2dWarmStart {
1166            x0: (0.5 * 2.77777777777777828e2, LP_BOUND),
1167            skip: 2,
1168        };
1169
1170        // The optimum the rows actually admit, confirmed independently in exact
1171        // rational arithmetic.
1172        let y_true = 2.82287152777735628e2;
1173
1174        let (x_incre, y_incre) = lp_2d_incre_max_y::<_, false>(&mut a_b, &warm_start, &tol);
1175        assert!(
1176            x_incre.is_finite() && y_incre.is_finite(),
1177            "the incremental solver reported a feasible problem as infeasible"
1178        );
1179        assert!(
1180            (x_incre - 2.77777777777777828e2).abs() < 1.0e-9 && (y_incre - y_true).abs() < 1.0e-9,
1181            "incremental ({x_incre}, {y_incre}) should match the true optimum y={y_true}"
1182        );
1183        assert!(
1184            a_b.iter()
1185                .all(|&(a, b, c)| a * x_incre + b * y_incre - c <= tol.feas_tol),
1186            "the returned point must satisfy every normalized row within feas_tol"
1187        );
1188
1189        // The slow recovery must also reconstruct a real supporting basis,
1190        // which is what the indexed API reports to its callers.
1191        let repair_epsilon = tol.feas_tol * LP2D_REPAIR_TOL_SCALE;
1192        let repair_tol_1d =
1193            LpToleranceOptions::with_feas_tol(repair_epsilon * tol.reduce_dim_scale);
1194        let mut active = IndexLpIncCollector::new();
1195        let (x_recovered, y_recovered) = recover_2d_prefix(
1196            &a_b,
1197            a_b.len(),
1198            x_incre,
1199            y_incre,
1200            repair_epsilon,
1201            &repair_tol_1d,
1202            &mut active,
1203            LP_BOUND,
1204        )
1205        .expect("the alternate active-boundary scan should recover this prefix");
1206        assert!((x_recovered - x_incre).abs() <= repair_epsilon);
1207        assert!((y_recovered - y_incre).abs() <= repair_epsilon);
1208        let (id0, id1) = (active.id.0, active.id.1);
1209        assert!(id0 < a_b.len() && id1 < a_b.len());
1210        let (n0, n1) = (a_b[id0], a_b[id1]);
1211        assert!((n0.0 * x_recovered + n0.1 * y_recovered - n0.2).abs() <= repair_epsilon);
1212        assert!((n1.0 * x_recovered + n1.1 * y_recovered - n1.2).abs() <= repair_epsilon);
1213        let det = n0.0 * n1.1 - n0.1 * n1.0;
1214        let (lambda0, lambda1) = (-n1.0 / det, n0.0 / det);
1215        assert!(
1216            det.abs() > CCW_ERRBOUND_A && lambda0 >= -repair_epsilon && lambda1 >= -repair_epsilon,
1217            "the recovered active normals must support the +y objective"
1218        );
1219    }

1221    /// The box the kernel poses its problem over is a starting point, not a
1222    /// limit: an answer on its edge is a clipped one, so the box grows and the
1223    /// solve repeats.
1224    ///
1225    /// The rows below admit `max y = 1e8` at `x = 0`, two orders of magnitude
1226    /// above [`LP_BOUND`]. The first solve can only reach the edge of the
1227    /// starting box and reports saturation; without the growth loop the answer
1228    /// would stay at `LP_BOUND`.
1229    #[test]
1230    fn test_lp_2d_incre_max_y_grows_the_box_past_its_start() {
1231        // x + y <= 1e8, -x + y <= 1e8, y >= 0.
1232        let mut a_b = vec![(1.0, 1.0, 1.0e8), (-1.0, 1.0, 1.0e8), (0.0, -1.0, 0.0)];
1233        normalize_lp2d(&mut a_b);
1234        let tol = LpToleranceOptions::with_feas_tol(1.0e-8);
1235        let warm_start = Lp2dWarmStart {
1236            x0: (0.0, LP_BOUND),
1237            skip: 0,
1238        };
1239
1240        let (x, y) = lp_2d_incre_max_y::<_, false>(&mut a_b, &warm_start, &tol);
1241        assert!(
1242            y > LP_BOUND,
1243            "the box never grew: y={y} is still the edge of the starting box"
1244        );
1245        assert!(
1246            x.abs() < 1.0 && (y - 1.0e8).abs() < 1.0,
1247            "expected the apex (0, 1e8), got ({x}, {y})"
1248        );
1249        // The rows carry unit normals but sit at a right-hand side of `1e8`, so
1250        // a residual is only meaningful against that scale, not against the
1251        // feasibility tolerance the solver was given.
1252        assert!(
1253            a_b.iter().all(|&(a, b, c)| a * x + b * y - c <= 1.0e-3),
1254            "the returned point must satisfy every row at the scale of its rows"
1255        );
1256    }
