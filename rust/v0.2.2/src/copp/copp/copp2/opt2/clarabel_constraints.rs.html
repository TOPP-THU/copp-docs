1//! Shared Clarabel constraint assembly for second-order path-parameterization backends.
2//!
3//! # Method identity
4//! This module is shared by:
5//! - **Time-Optimal Path Parameterization (TOPP2)** solvers,
6//! - **Convex-Objective Path Parameterization (COPP2)** solvers.
7//!
8//! # Scope
9//! This module contains reusable routines that convert second-order path constraints into
10//! Clarabel triplet form `(row, col, val, b, cones)`.
11//!
12//! # Constraint convention
13//! Constraints are assembled in Clarabel form:
14//! - `s = b - A*x in K`
15//! - equivalent implementation view: `A*x - b = -s`.
16
17use crate::copp::clarabel_backend::ConstraintsClarabel;
18use crate::copp::constraints::Constraints;
19use crate::copp::copp2::formulation::Topp2Problem;
20use crate::diag::{Verboser, Verbosity};
21use clarabel::solver::{NonnegativeConeT, SupportedConeT, ZeroConeT};
22use core::f64;

24/// Assemble the standard TOPP2 feasibility constraints into Clarabel triplet buffers.
25///
26/// # Included blocks
27/// 1. Boundary equalities: `a[0] = a_start`, `a[n] = a_final`.
28/// 2. First-order bounds for interior nodes.
29/// 3. Second-order acceleration bounds using finite-difference coupling.
30/// 4. Final nonnegative cone packing for all inequality rows added after boundary equalities.
31///
32/// # Logging behavior
33/// - [`Debug`](crate::diag::Verbosity::Debug): emits stage-by-stage matrix/vector/cone sizes.
34/// - [`Trace`](crate::diag::Verbosity::Trace): no extra output in this function yet; Trace currently inherits Debug visibility.
35pub(crate) fn clarabel_standard_constraint_topp2(
36    problem: &Topp2Problem,
37    constraints: ConstraintsClarabel,
38    verboser: &impl Verboser,
39) {
40    let (idx_s_start, idx_s_final) = problem.idx_s_interval;
41    let (a_start, a_final) = problem.a_boundary;
42    let n = idx_s_final - idx_s_start;
43    let (row, col, val, b, cones) = constraints;
44
45    let report_counts = |phase: &str,
46                         row: &Vec<usize>,
47                         col: &Vec<usize>,
48                         val: &Vec<f64>,
49                         b: &Vec<f64>,
50                         cones: &Vec<SupportedConeT<f64>>| {
51        if verboser.is_enabled(Verbosity::Debug) {
52            crate::verbosity_log!(
53                crate::diag::Verbosity::Summary,
54                "clarabel_standard_constraint_topp2[{phase}]: row={}, col={}, val={}, b={}, cones={}",
55                row.len(),
56                col.len(),
57                val.len(),
58                b.len(),
59                cones.len()
60            );
61        }
62    };
63
64    // Step 1.1 initial constraints (num_val<=2, num_b<=2, num_cone<=1)
65    clarabel_a_point_constraint_topp2(a_start, 0, (row, col, val, b, cones));
66    report_counts("a_start", row, col, val, b, cones);
67    // Step 1.2 terminal constraints (num_val<=2, num_b<=2, num_cone<=1)
68    clarabel_a_point_constraint_topp2(a_final, n, (row, col, val, b, cones));
69    report_counts("a_final", row, col, val, b, cones);
70    let n_rows_old = b.len();
71    // Step 1.3 first-order constraints (num_val<=2*(n-1), num_b<=2*(n-1), num_cone<=1 if packed independently)
72    // A*x-b = -s = -1*a[k] <= 0
73    // A*x-b = -s = 1*a[k] - amax[k] <= 0
74    clarabel_1order_constraint_topp2(
75        problem.constraints,
76        n,
77        idx_s_start,
78        (row, col, val, b, cones),
79        false,
80    );
81    report_counts("first_order", row, col, val, b, cones);
82    // Step 1.4 second-order constraints (num_val<=4*count_acc_constraints, num_b<=2*count_acc_constraints, num_cone<=1 if packed independently)
83    // acc_a * a[k] + acc_b * b[k-1] <= acc_max
84    // acc_a * a[k] + acc_b * b[k] <= acc_max
85    clarabel_2order_constraint_topp2(
86        problem.constraints,
87        idx_s_start,
88        n,
89        (row, col, val, b, cones),
90        false,
91    );
92    report_counts("second_order", row, col, val, b, cones);
93    cones.push(NonnegativeConeT(b.len() - n_rows_old));
94    report_counts("final_nonnegative_cone", row, col, val, b, cones);
95}

97/// Estimate allocation capacities for standard TOPP2 feasibility constraints.
98///
99/// Returns `(capacity_val, capacity_b, capacity_cones)` where:
100/// - `capacity_val`: estimated nonzero triplets in `A`;
101/// - `capacity_b`: estimated row count (`b.len()`);
102/// - `capacity_cones`: cone blocks count (boundary zero-cones + inequality nonnegative-cone).
103#[inline]
104pub(crate) fn clarabel_standard_capacity_topp2(
105    constraints: &Constraints,
106    idx_s_interval: (usize, usize),
107) -> (usize, usize, usize) {
108    let (idx_s_start, idx_s_final) = idx_s_interval;
109    let n = idx_s_final - idx_s_start;
110    let count_acc_constraints = constraints.count_rows_acc(idx_s_start, idx_s_final + 1);
111    // Step 1. Boundary constraints (num_val<=4; num_b<=4; num_cone==2)
112    // Step 2. first-order constraints
113    // num_val<=2*(n-1), num_b<=2*(n-1)
114    // Step 3. second-order constraints
115    // num_val<=4*count_acc_constraints, num_b<=2*count_acc_constraints
116    // Step 4. final inequality cone packing (num_cone==1)
117    let capacity_val = 4 * count_acc_constraints + 2 * (n + 1);
118    let capacity_b = 2 * count_acc_constraints + 2 * (n + 1);
119    (capacity_val, capacity_b, 3)
120}

122/// Add one boundary equality `a[id_col] = a_boundary` as a one-row Clarabel [`ZeroConeT`](clarabel::solver::SupportedConeT::ZeroConeT).
123///
124/// Note: this function always appends data and does not perform feasibility checks.
125fn clarabel_a_point_constraint_topp2(
126    a_boundary: f64,
127    id_col: usize,
128    constraints: ConstraintsClarabel,
129) {
130    let (row, col, val, b, cones) = constraints;
131    // A*x-b = -s = 1*a[id_col] - a = 0
132    row.push(b.len());
133    col.push(id_col);
134    val.push(1.0);
135    b.push(a_boundary);
136    cones.push(ZeroConeT(1));
137}

139/// Append first-order bounds for interior samples `k=1..n-1`.
140///
141/// Added inequalities:
142/// - lower bound: `a[k] >= 0`;
143/// - upper bound: `a[k] <= amax[k]` (only when `amax[k]` is finite).
144///
145/// Upper-bound estimate: `num_val<=2*(n-1)`, `num_b<=2*(n-1)`, `num_cone<=1`.
146///
147/// If `modify_cones=true`, a single [`NonnegativeConeT`](clarabel::solver::SupportedConeT::NonnegativeConeT) is appended for all added rows.
148fn clarabel_1order_constraint_topp2(
149    constraints: &Constraints,
150    n: usize,
151    idx_s_start: usize,
152    input: ConstraintsClarabel,
153    modify_cones: bool,
154) {
155    let (row, col, val, b, cones) = input;
156    // A*x-b = -s = -1*a[k] <= 0
157    row.extend((b.len() + 1)..(b.len() + n));
158    col.extend(1..n);
159    val.resize(val.len() + n - 1, -1.0);
160    b.resize(b.len() + n - 1, 0.0);
161    // A*x-b = -s = 1*a[k] - amax[k] <= 0
162    let amax_constraint: Vec<(usize, f64)> = (1..n)
163        .filter_map(|k| {
164            let amax_curr = constraints.amax_unchecked(idx_s_start + k);
165            if amax_curr.is_finite() {
166                Some((k, amax_curr))
167            } else {
168                None
169            }
170        })
171        .collect();
172    row.extend(b.len()..(b.len() + amax_constraint.len()));
173    col.extend(amax_constraint.iter().map(|(col_, _)| *col_));
174    val.resize(val.len() + amax_constraint.len(), 1.0);
175    b.extend(amax_constraint.iter().map(|(_, b_)| *b_));
176    if modify_cones {
177        cones.push(NonnegativeConeT(n - 1 + amax_constraint.len()));
178    }
179}

181/// Append second-order acceleration constraints across all stages `k=0..n`.
182///
183/// This routine converts constraints involving `(a[k], b[k-1])` / `(a[k], b[k])` into
184/// linear inequalities over adjacent `a` entries via finite-difference substitutions.
185///
186/// Upper-bound estimate: `num_val<=4*count_acc_constraints`,
187/// `num_b<=2*count_acc_constraints`, `num_cone<=1`.
188///
189/// If `modify_cones=true`, a single [`NonnegativeConeT`](clarabel::solver::SupportedConeT::NonnegativeConeT) is appended for all added rows.
190fn clarabel_2order_constraint_topp2(
191    constraints: &Constraints,
192    idx_s_start: usize,
193    n: usize,
194    input: ConstraintsClarabel,
195    modify_cones: bool,
196) {
197    let (row, col, val, b, cones) = input;
198    let n_b_old = b.len();
199    for k in 0..=n {
200        let (acc_a, acc_b, acc_max) = constraints.acc_constraints_unchecked(idx_s_start + k);
201        let nrows = acc_a.nrows();
202        if k > 0 {
203            // acc_a * a[k] + acc_b * b[k-1] <= acc_max
204            // acc_a * a[k] + acc_b * (a[k] - a[k-1]) / ds_double <= acc_max
205            // A*x-b = -s = (acc_a + acc_b / ds_double) * a[k] - acc_b / ds_double * a[k-1] - acc_max <= 0
206            let ds_double_reciprocal = 0.5
207                / (constraints.s_unchecked(idx_s_start + k)
208                    - constraints.s_unchecked(idx_s_start + k - 1));
209            row.extend(b.len()..(b.len() + nrows));
210            row.extend(b.len()..(b.len() + nrows));
211            col.resize(col.len() + nrows, k - 1);
212            col.resize(col.len() + nrows, k);
213            val.extend(acc_b.iter().map(|&b_coeff| -b_coeff * ds_double_reciprocal));
214            val.extend(
215                acc_a
216                    .iter()
217                    .zip(acc_b.iter())
218                    .map(|(&a_coeff, &b_coeff)| a_coeff + b_coeff * ds_double_reciprocal),
219            );
220            b.extend(acc_max.iter());
221        }
222        if k < n {
223            // acc_a * a[k] + acc_b * b[k] <= acc_max
224            // acc_a * a[k] + acc_b * (a[k+1] - a[k]) / ds_double <= acc_max
225            // A*x-b = -s = (acc_a - acc_b / ds_double) * a[k] + acc_b / ds_double * a[k+1] - acc_max <= 0
226            let ds_double_reciprocal = 0.5
227                / (constraints.s_unchecked(idx_s_start + k + 1)
228                    - constraints.s_unchecked(idx_s_start + k));
229            row.extend(b.len()..(b.len() + nrows));
230            row.extend(b.len()..(b.len() + nrows));
231            col.resize(col.len() + nrows, k);
232            col.resize(col.len() + nrows, k + 1);
233            val.extend(
234                acc_a
235                    .iter()
236                    .zip(acc_b.iter())
237                    .map(|(&a_coeff, &b_coeff)| a_coeff - b_coeff * ds_double_reciprocal),
238            );
239            val.extend(acc_b.iter().map(|&b_coeff| b_coeff * ds_double_reciprocal));
240            b.extend(acc_max.iter());
241        }
242    }
243    if modify_cones {
244        cones.push(NonnegativeConeT(b.len() - n_b_old));
245    }
