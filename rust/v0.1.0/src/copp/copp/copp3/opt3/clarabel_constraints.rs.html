1//! Shared Clarabel constraint assembly for third-order path-parameterization backends.
2//!
3//! # Method identity
4//! This module is shared by:
5//! - **Time-Optimal Path Parameterization (TOPP3)** solvers,
6//! - **Convex-Objective Path Parameterization (COPP3)** solvers.
7//!
8//! # Scope
9//! This module contains reusable routines that convert third-order path constraints into
10//! Clarabel triplet form `(row, col, val, b, cones)`.
11//!
12//! # Constraint convention
13//! Constraints are assembled in Clarabel form:
14//! - `s = b - A*x in K`
15//! - equivalent implementation view: `A*x - b = -s`.
16
17use crate::copp::clarabel_backend::ConstraintsClarabel;
18use crate::copp::constraints::Constraints;
19use crate::copp::copp3::formulation::Topp3Problem;
20use crate::diag::{CoppError, Verboser, Verbosity};
21use clarabel::solver::{NonnegativeConeT, SupportedConeT, ZeroConeT};
22use core::f64;

24/// Assemble the standard TOPP3 feasibility constraints into Clarabel triplet buffers.
25///
26/// # Included blocks
27/// 1. Boundary equalities for `a` and `b` at start/end samples.
28/// 2. Dynamic equalities for stationary and moving intervals.
29/// 3. First-order bounds (including stationary-endpoint handling).
30/// 4. Second-order acceleration bounds.
31/// 5. Third-order jerk-linear bounds.
32/// 6. Final nonnegative cone packing for all inequality rows added after dynamic equalities.
33///
34/// # Logging behavior
35/// - `Debug`: emits stage-by-stage matrix/vector/cone sizes.
36/// - `Trace`: no extra output in this function yet; Trace currently inherits Debug visibility.
37pub(crate) fn clarabel_standard_constraint_topp3(
38    problem: &Topp3Problem,
39    s: &[f64],
40    constraints: ConstraintsClarabel,
41    num_stationary: (usize, usize),
42    verboser: &impl Verboser,
43) -> Result<(), CoppError> {
44    let n = problem.a_linearization.len() - 1;
45    // s=b-A*x \in cone, where A[row[i],col[i]]=val[i], A \in R^{m*(n+1)}, b \in R^m, s \in R^m
46    // -s=-b+A*x
47    let (row, col, val, b, cones) = constraints;
48    let report_counts = |phase: &str,
49                         row: &Vec<usize>,
50                         col: &Vec<usize>,
51                         val: &Vec<f64>,
52                         b: &Vec<f64>,
53                         cones: &Vec<SupportedConeT<f64>>| {
54        if verboser.is_enabled(Verbosity::Debug) {
55            crate::verbosity_log!(
56                crate::diag::Verbosity::Summary,
57                "clarabel_standard_constraint_topp3[{phase}]: row={}, col={}, val={}, b={}, cones={}",
58                row.len(),
59                col.len(),
60                val.len(),
61                b.len(),
62                cones.len()
63            );
64        }
65    };
66    // Step 1. Boundary constraints (num_val==4; num_b==4; num_cone==0 here)
67    clarabel_boundary_constraint_topp3(
68        n,
69        problem.a_boundary,
70        problem.b_boundary,
71        (row, col, val, b, cones),
72        false,
73    );
74    report_counts("boundary", row, col, val, b, cones);
75    // Step 2. Dynamic constraints (num_val<=4*n; num_b<=n; num_cone==1 after dynamic cone packing)
76    clarabel_dynamic_constraint_topp3(s, num_stationary, (row, col, val, b, cones), false);
77    cones.push(ZeroConeT(b.len()));
78    report_counts("dynamic+zero_cone", row, col, val, b, cones);
79    let n_b_old = b.len();
80    // Step 3. first-order constraints
81    // num_val<=2*(n-1), num_b<=2*(n-1), num_cone<=1 (if packed independently)
82    if !clarabel_1order_constraint_topp3(
83        problem,
84        s,
85        num_stationary,
86        (row, col, val, b, cones),
87        false,
88    ) {
89        return Err(CoppError::Infeasible(
90            "clarabel_standard_constraint_topp3".into(),
91            "Failed to add first-order constraints.".into(),
92        ));
93    }
94    report_counts("first_order", row, col, val, b, cones);
95    // Step 4. second-order constraints
96    // num_val<=2*count_acc_constraints, num_b<=count_acc_constraints, num_cone<=1 (if packed independently)
97    clarabel_2order_constraint_topp3(problem, s, num_stationary, (row, col, val, b, cones), false);
98    report_counts("second_order", row, col, val, b, cones);
99    // Step 5. third-order constraints
100    // num_val<=6*count_jerk_constraints, num_b<=2*count_jerk_constraints, num_cone<=1 (if packed independently)
101    clarabel_3order_constraint_topp3(problem, s, num_stationary, (row, col, val, b, cones), false);
102    cones.push(NonnegativeConeT(b.len() - n_b_old));
103    report_counts("third_order+final_nonnegative", row, col, val, b, cones);
104    Ok(())
105}

107/// Append third-order jerk-linear constraints.
108///
109/// Added inequalities are generated per stage from
110/// `jerk_a_linear * a + jerk_b * b + jerk_c * c <= jerk_max_linear`
111/// using the finite-difference coupling between `c` and adjacent `b`.
112///
113/// Upper-bound estimate: `num_val<=6*count_jerk_constraints`,
114/// `num_b<=2*count_jerk_constraints`, `num_cone<=1`.
115///
116/// If `modify_cones=true`, a single `NonnegativeConeT` is appended for all added rows.
117fn clarabel_3order_constraint_topp3(
118    problem: &Topp3Problem,
119    s: &[f64],
120    num_stationary: (usize, usize),
121    constraints: ConstraintsClarabel,
122    modify_cones: bool,
123) {
124    let (row, col, val, b, cones) = constraints;
125    let n_b_old = b.len();
126    let n = s.len() - 1;
127
128    for k in (num_stationary.0)..=(n - num_stationary.1) {
129        let (jerk_a_linear, jerk_b, jerk_c, jerk_max_linear) = problem
130            .constraints
131            .jerk_linear_constraints_unchecked(problem.idx_s_start + k);
132        let nrows = jerk_a_linear.nrows();
133
134        if k > num_stationary.0 {
135            // jerk_a_linear * a[k] + jerk_b * b[k] + jerk_c * c[k-1] <= jerk_max_linear
136            // jerk_a_linear * a[k] + jerk_b * b[k] + jerk_c / ds * (b[k] - b[k-1]) <= jerk_max_linear
137            // jerk_a_linear * a[k] - jerk_c / ds * b[k-1] + (jerk_b + jerk_c / ds) * b[k] <= jerk_max_linear
138            // A*x-b = -s = jerk_a_linear * x[k] - jerk_ds_down * x[n+k] + (jerk_b + jerk_ds_down) * b[n+k+1] - jerk_max_linear <= 0
139            let ds_down = 1.0 / (s[k] - s[k - 1]);
140            row.extend(b.len()..(b.len() + nrows));
141            row.extend(b.len()..(b.len() + nrows));
142            row.extend(b.len()..(b.len() + nrows));
143            col.resize(col.len() + nrows, k);
144            col.resize(col.len() + nrows, n + k);
145            col.resize(col.len() + nrows, n + k + 1);
146            val.extend(jerk_a_linear.iter());
147            val.extend(jerk_c.iter().map(|&c_coeff| -c_coeff * ds_down));
148            val.extend(
149                jerk_b
150                    .iter()
151                    .zip(jerk_c.iter())
152                    .map(|(&b_coeff, &c_coeff)| b_coeff + c_coeff * ds_down),
153            );
154            b.extend(jerk_max_linear.iter());
155        }
156
157        if k < n - num_stationary.1 {
158            // jerk_a_linear * a[k] + jerk_b * b[k] + jerk_c * c[k] <= jerk_max_linear
159            // jerk_a_linear * a[k] + jerk_b * b[k] + jerk_c / ds * (b[k+1] - b[k]) <= jerk_max_linear
160            // jerk_a_linear * a[k] + (jerk_b - jerk_c / ds) * b[k] + jerk_c / ds * b[k+1] <= jerk_max_linear
161            // A*x-b = -s = jerk_a_linear * x[k] + (jerk_b - jerk_c / ds) * x[n+k+1] + jerk_c / ds * x[n+k+2] - jerk_max_linear <= 0
162            let ds_down = 1.0 / (s[k + 1] - s[k]);
163            row.extend(b.len()..(b.len() + nrows));
164            row.extend(b.len()..(b.len() + nrows));
165            row.extend(b.len()..(b.len() + nrows));
166            col.resize(col.len() + nrows, k);
167            col.resize(col.len() + nrows, n + k + 1);
168            col.resize(col.len() + nrows, n + k + 2);
169            val.extend(jerk_a_linear.iter());
170            val.extend(
171                jerk_b
172                    .iter()
173                    .zip(jerk_c.iter())
174                    .map(|(&b_coeff, &c_coeff)| b_coeff - c_coeff * ds_down),
175            );
176            val.extend(jerk_c.iter().map(|&c_coeff| c_coeff * ds_down));
177            b.extend(jerk_max_linear.iter());
178        }
179    }
180
181    if modify_cones {
182        cones.push(NonnegativeConeT(b.len() - n_b_old));
183    }
184}

186/// Append second-order acceleration constraints on moving intervals.
187///
188/// Added inequalities follow
189/// `acc_a * a[k] + acc_b * b[k] <= acc_max`.
190///
191/// Upper-bound estimate: `num_val<=2*count_acc_constraints`,
192/// `num_b<=count_acc_constraints`, `num_cone<=1`.
193///
194/// If `modify_cones=true`, a single `NonnegativeConeT` is appended for all added rows.
195fn clarabel_2order_constraint_topp3(
196    problem: &Topp3Problem,
197    s: &[f64],
198    num_stationary: (usize, usize),
199    constraints: ConstraintsClarabel,
200    modify_cones: bool,
201) {
202    let (row, col, val, b, cones) = constraints;
203    let n_b_old = b.len();
204    let n = s.len() - 1;
205    for k in (num_stationary.0 + 1)..(n - num_stationary.1) {
206        let (acc_a, acc_b, acc_max) = problem
207            .constraints
208            .acc_constraints_unchecked(problem.idx_s_start + k);
209        let nrows = acc_a.nrows();
210        // acc_a * a[k] + acc_b * b[k] <= acc_max
211        // A*x-b = -s = acc_a * a[k] + acc_b * b[k] - acc_max = acc_a * x[k] + acc_b * x[n+k+1] - acc_max <= 0
212        row.extend(b.len()..(b.len() + nrows));
213        row.extend(b.len()..(b.len() + nrows));
214        col.resize(col.len() + nrows, k);
215        col.resize(col.len() + nrows, n + k + 1);
216        val.extend(acc_a.iter());
217        val.extend(acc_b.iter());
218        b.extend(acc_max.iter());
219    }
220    if modify_cones {
221        cones.push(NonnegativeConeT(b.len() - n_b_old));
222    }
223}

225/// Append first-order bounds with stationary-endpoint handling.
226///
227/// Added inequalities include:
228/// - stationary-endpoint bounds (when stationary segments exist at start/end),
229/// - interior lower bounds `a[k] >= 0`,
230/// - interior upper bounds `a[k] <= amax[k]` (only when `amax[k]` is finite).
231///
232/// Upper-bound estimate: `num_val<=2*(n-1)`, `num_b<=2*(n-1)`, `num_cone<=1`.
233///
234/// Returns `false` when stationary-endpoint bounds are infeasible.
235///
236/// If `modify_cones=true`, a single `NonnegativeConeT` is appended for all added rows.
237fn clarabel_1order_constraint_topp3(
238    problem: &Topp3Problem,
239    s: &[f64],
240    num_stationary: (usize, usize),
241    constraints: ConstraintsClarabel,
242    modify_cones: bool,
243) -> bool {
244    let (row, col, val, b, cones) = constraints;
245    let n_b_old = b.len();
246    let n = s.len() - 1;
247    // num_val<=2*num_stationary.0; num_b<=2*num_stationary.0
248    if num_stationary.0 > 0 {
249        let (amax_stationary, amin_stationary) =
250            problem.constraints.stationary_constraint_topp3::<false>(
251                problem.a_linearization,
252                problem.idx_s_start,
253                num_stationary.0,
254            );
255        // amin_stationary <= a[num_stationary.0] <= amax_stationary
256        if amax_stationary < amin_stationary {
257            // Infeasible
258            return false;
259        }
260        if amax_stationary.is_finite() {
261            // A*x-b = -s = 1*a[num_stationary.0] - amax_stationary = 1*x[num_stationary.0] - amax_stationary <= 0
262            row.push(b.len());
263            col.push(num_stationary.0);
264            val.push(1.0);
265            b.push(amax_stationary);
266        }
267        // A*x-b = -s = -1*a[num_stationary.0] + amin_stationary = -1*x[num_stationary.0] + amin_stationary <= 0
268        row.push(b.len());
269        col.push(num_stationary.0);
270        val.push(-1.0);
271        b.push(-amin_stationary);
272    }
273    // num_val<=2*num_stationary.0; num_b<=2*num_stationary.0
274    if num_stationary.1 > 0 {
275        let (amax_stationary, amin_stationary) =
276            problem.constraints.stationary_constraint_topp3::<true>(
277                problem.a_linearization,
278                problem.idx_s_start,
279                num_stationary.1,
280            );
281        // amin_stationary <= a[n - num_stationary.1] <= amax_stationary
282        if amax_stationary < amin_stationary {
283            // Infeasible
284            return false;
285        }
286        let id_a = n - num_stationary.1;
287        if amax_stationary.is_finite() {
288            // A*x-b = -s = 1*a[n - num_stationary.1] - amax_stationary = 1*x[id_a] - amax_stationary <= 0
289            row.push(b.len());
290            col.push(id_a);
291            val.push(1.0);
292            b.push(amax_stationary);
293        }
294        // A*x-b = -s = -1*a[n - num_stationary.1] + amin_stationary = -1*x[id_a] + amin_stationary <= 0
295        row.push(b.len());
296        col.push(id_a);
297        val.push(-1.0);
298        b.push(-amin_stationary);
299    }
300    // 0<= a[k] <= amax[k] for k in (num_stationary.0+1) .. (n - num_stationary.1)
301    // A*x-b = -s = 1*a[k] - amax[k] = 1*x[k] - amax[k] <= 0
302    // num_val<=2*(n - num_stationary.0 - num_stationary.1 - 1)
303    // num_b<=2*(n - num_stationary.0 - num_stationary.1 - 1)
304    // A*x-b = -s = -1*a[k] <= 0
305    let n_b_here = n - num_stationary.1 - num_stationary.0 - 1;
306    row.extend(b.len()..(b.len() + n_b_here));
307    col.extend((num_stationary.0 + 1)..(n - num_stationary.1));
308    val.resize(val.len() + n_b_here, -1.0);
309    b.resize(b.len() + n_b_here, 0.0);
310    // A*x-b = -s = 1*a[k] - amax[k] <= 0
311    let amax_constraint: Vec<(usize, f64)> = ((num_stationary.0 + 1)..(n - num_stationary.1))
312        .filter_map(|k| {
313            let amax_curr = problem.constraints.amax_unchecked(problem.idx_s_start + k);
314            if amax_curr.is_finite() {
315                Some((k, amax_curr))
316            } else {
317                None
318            }
319        })
320        .collect();
321    row.extend(b.len()..(b.len() + amax_constraint.len()));
322    col.extend(amax_constraint.iter().map(|(col_, _)| *col_));
323    val.resize(val.len() + amax_constraint.len(), 1.0);
324    b.extend(amax_constraint.iter().map(|(_, b_)| *b_));
325    // clarabel_1order_constraint_topp2(
326    //     constraints,
327    //     idx_s_start + num_stationary.0,
328    //     n - num_stationary.0 - num_stationary.1,
329    //     (row, col, val, b, cones),
330    //     false,
331    // );
332
333    if modify_cones {
334        cones.push(NonnegativeConeT(b.len() - n_b_old));
335    }
336    true
337}

339/// Append boundary equalities for `a` and `b` as Clarabel zero-cone rows.
340///
341/// Upper-bound estimate: `num_val==4`, `num_b==4`, `num_cone<=1`.
342///
343/// If `modify_cones=true`, appends `ZeroConeT(4)` for the newly added rows.
344fn clarabel_boundary_constraint_topp3(
345    n: usize,
346    a_boundary: (f64, f64),
347    b_boundary: (f64, f64),
348    constraints: ConstraintsClarabel,
349    modify_cones: bool,
350) {
351    let (row, col, val, b, cones) = constraints;
352    // A*x-b = -s = 1*a[0]-a_boundary.0 = 1*x[0]-a_boundary.0 = 0
353    // A*x-b = -s = 1*a[n]-a_boundary.1 = 1*x[n]-a_boundary.1 = 0
354    // A*x-b = -s = 1*b[0]-b_boundary.0 = 1*x[n+1]-b_boundary.0 = 0
355    // A*x-b = -s = 1*b[n]-b_boundary.1 = 1*x[2*n+1]-b_boundary.1 = 0
356    row.extend(b.len()..(b.len() + 4));
357    col.extend([0, n, n + 1, 2 * n + 1]);
358    val.resize(val.len() + 4, 1.0);
359    b.extend([a_boundary.0, a_boundary.1, b_boundary.0, b_boundary.1]);
360    if modify_cones {
361        cones.push(ZeroConeT(4));
362    }
363}

365/// Append dynamic equalities for both stationary and moving intervals.
366///
367/// Upper-bound estimate: `num_val<=4*n`, `num_b<=n`, `num_cone<=1`.
368///
369/// If `modify_cones=true`, appends one `ZeroConeT` that covers all rows added by this routine.
370fn clarabel_dynamic_constraint_topp3(
371    s: &[f64],
372    num_stationary: (usize, usize),
373    constraints: ConstraintsClarabel,
374    modify_cones: bool,
375) {
376    let (row, col, val, b, cones) = constraints;
377    let n_b_old = b.len();
378    if num_stationary.0 > 0 {
379        // num_val==2; num_b==1
380        clarabel_dynamic_stationary_topp3(s, num_stationary.0, false, (row, col, val, b, cones));
381    }
382    if num_stationary.1 > 0 {
383        // num_val==2; num_b==1
384        clarabel_dynamic_stationary_topp3(s, num_stationary.1, true, (row, col, val, b, cones));
385    }
386    // num_val==4*(n-num_stationary.0-num_stationary.1)
387    // num_b==(n - num_stationary.0 - num_stationary.1)
388    clarabel_dynamic_moving_topp3(s, num_stationary, (row, col, val, b, cones));
389    if modify_cones {
390        cones.push(ZeroConeT(b.len() - n_b_old));
391    }
392}

394/// Append dynamic equality for one stationary boundary interval.
395///
396/// Upper-bound estimate: `num_val==2`, `num_b==1`, `num_cone==0`.
397fn clarabel_dynamic_stationary_topp3(
398    s: &[f64],
399    num_stationary: usize,
400    reverse: bool,
401    constraints: ConstraintsClarabel,
402) {
403    let (row, col, val, b, _) = constraints;
404    let n = s.len() - 1;
405    // 3*(s[idx_s_start - num_stationary] - s[idx_s_start])* b[n - num_stationary] == 2* a[n - num_stationary] (reverse==true)
406    // 3*(s[idx_s_start + num_stationary] - s[idx_s_start])* b[num_stationary] == 2* a[num_stationary] (reverse==false)
407    let (id_a, s_start) = if reverse {
408        (n - num_stationary, s.last().unwrap())
409    } else {
410        (num_stationary, s.first().unwrap())
411    };
412    // a[num_stationary] - 1.5*(s[id_a] - s_start)* b[num_stationary] == 0
413    let coeff = -1.5 * (s[id_a] - s_start);
414    // A*x-b = -s = 1*a[id_a] + coeff*b[id_a] = 1*x[id_a] + coeff*x[n+id_a+1] = 0
415    row.resize(row.len() + 2, b.len());
416    col.extend([id_a, n + 1 + id_a]);
417    val.extend([1.0, coeff]);
418    b.push(0.0);
419}

421/// Append dynamic equalities for moving intervals.
422///
423/// This routine encodes the finite-difference identity
424/// `a[k] - a[k+1] + ds[k]*b[k] + ds[k]*b[k+1] = 0`.
425///
426/// Upper-bound estimate: `num_val==4*(n-num_stationary.0-num_stationary.1)`,
427/// `num_b==(n-num_stationary.0-num_stationary.1)`, `num_cone==0`.
428fn clarabel_dynamic_moving_topp3(
429    s: &[f64],
430    num_stationary: (usize, usize),
431    constraints: ConstraintsClarabel,
432) {
433    let (row, col, val, b, _) = constraints;
434    // b[k+1] - b[k] = c[k] * ds[k]
435    // a[k+1] - a[k] = 2.0 * b[k] * ds[k] + c[k] * ds[k]^2 = 2.0 * b[k] * ds[k] + (b[k+1]-b[k]) * ds[k]
436    // a[k] - a[k+1] + ds[k]*b[k] + ds[k]*b[k+1] = 0
437    // A*x-b = -s = x[k] - x[k+1] + ds[k]*x[n+k+1] + ds[k]*b[n+k+2] = 0
438    // for k in num_stationary.0 .. n - num_stationary.1
439    let n = s.len() - 1;
440    let n_b_here = n - num_stationary.0 - num_stationary.1;
441    row.extend(b.len()..(b.len() + n_b_here));
442    row.extend(b.len()..(b.len() + n_b_here));
443    row.extend(b.len()..(b.len() + n_b_here));
444    row.extend(b.len()..(b.len() + n_b_here));
445    col.extend(num_stationary.0..(num_stationary.0 + n_b_here)); // < n - num_stationary.1
446    col.extend((num_stationary.0 + 1)..(num_stationary.0 + n_b_here + 1)); // < n - num_stationary.1 + 1
447    col.extend((n + num_stationary.0 + 1)..(n + num_stationary.0 + n_b_here + 1)); // < 2 * n - num_stationary.1
448    col.extend((n + num_stationary.0 + 2)..(n + num_stationary.0 + n_b_here + 2)); // < 2 * n - num_stationary.1 + 1
449    val.resize(val.len() + n_b_here, 1.0);
450    val.resize(val.len() + n_b_here, -1.0);
451    val.extend(
452        s.windows(2)
453            .skip(num_stationary.0)
454            .take(n_b_here)
455            .map(|s_pair| s_pair[1] - s_pair[0]),
456    );
457    val.extend(
458        s.windows(2)
459            .skip(num_stationary.0)
460            .take(n_b_here)
461            .map(|s_pair| s_pair[1] - s_pair[0]),
462    );
463    b.resize(b.len() + n_b_here, 0.0);
464}

466/// Estimate allocation capacities for standard TOPP3 feasibility constraints.
467///
468/// Returns `(capacity_val, capacity_b, capacity_cones)` where:
469/// - `capacity_val`: estimated nonzero triplets in `A`;
470/// - `capacity_b`: estimated row count (`b.len()`);
471/// - `capacity_cones`: estimated cone-block count.
472#[inline]
473pub(crate) fn clarabel_standard_capacity_topp3(
474    constraints: &Constraints,
475    idx_s_interval: (usize, usize),
476) -> (usize, usize, usize) {
477    let (idx_s_start, idx_s_final) = idx_s_interval;
478    let n = idx_s_final - idx_s_start;
479    let count_acc_constraints = constraints.count_rows_acc(idx_s_start, idx_s_final + 1);
480    let count_jerk_constraints = constraints.count_rows_jerk(idx_s_start, idx_s_final + 1);
481    // Step 1. Boundary constraints (num_val==4; num_b==4)
482    // Step 2. Dynamic constraints (num_val<=4*n; num_b<=n)
483    // Step 3. first-order constraints
484    // num_val<=2*(n-1), num_b<=2*(n-1)
485    // Step 4. second-order constraints
486    // num_val<=2*count_acc_constraints, num_b<=count_acc_constraints
487    // Step 5. third-order constraints
488    // num_val<=6*count_jerk_constraints, num_b<=2*count_jerk_constraints
489    // Step 6. cone packing
490    // num_cone==2 (1 x ZeroCone for equalities + 1 x NonnegativeCone for inequalities)
491    let capacity_val = 2 + 6 * n + 2 * count_acc_constraints + 6 * count_jerk_constraints;
492    let capacity_b = 2 + 3 * n + count_acc_constraints + 2 * count_jerk_constraints;
493    (capacity_val, capacity_b, 2)
494}
