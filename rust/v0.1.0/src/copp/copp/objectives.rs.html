1//! Objective definitions and shared validators for COPP2/COPP3.  
2//! Basic symbols (consistent with constraints):  
3//! - $a(s)=\dot s^2$, sampled as `a[k]=a(s_k)`.  
4//! - $b(s)=\ddot s$.  
5//! - Path grid is `s[0], s[1], ..., s[n-1]` with `n = s.len()`.
6//!
7//! Discretization difference:  
8//! - COPP2:  
9//!   - `a.len() == s.len() == n`  
10//!   - `b.len() == n-1`, where `b[k]` is on interval `[s_k, s_{k+1}]`  
11//!   - typically `b[k] = (a[k+1]-a[k]) / (2*(s[k+1]-s[k]))`  
12//! - COPP3:  
13//!   - `a.len() == b.len() == s.len() == n`  
14//!   - `b[k] = b(s_k)` is node-based.
15
16use crate::diag::CoppError;

18/// Objective terms for COPP optimization.  
19/// Continuous formulation is shared by COPP2/COPP3; discrete form depends on how `b` and torque are sampled.  
20/// Torque notation:  
21/// - continuous: $\boldsymbol{\tau}(s)$.  
22/// - discrete: `tau[i][k]` for joint `i` at `s[k]`.  
23/// - COPP2: `tau[i][k]` is the right-limit value $\boldsymbol{\tau}(s_k^+)$, i.e. computed on interval $[s_k, s_{k+1}]$ from `(a[k], a[k+1], b[k])`.  
24/// - COPP3: `tau[i][k]` is node value $\boldsymbol{\tau}(s_k)$, computed from `(a[k], b[k])`.
25pub enum CoppObjective<'a> {
26    /// Time objective.  
27    /// + Continuous: $J_{\mathrm{time}} = w_t\int_{0}^{t_f} 1 \mathrm{d}t = w_t\int_{s_0}^{s_f} \frac{1}{\sqrt{a(s)}} \mathrm{d}s$.  
28    /// + Discrete (COPP2): `J_time = 2*w_t*sum_{k=0}^{n-2} (s[k+1]-s[k])/(sqrt(a[k])+sqrt(a[k+1]))`.  
29    /// + Discrete (COPP3): `J_time = w_t*sum_k weight_a_time[k]/sqrt(a[k])`
30    Time(f64),
31    /// Thermal-energy objective.  
32    /// + Continuous: $J_{\mathrm{th}} = w_e\int_{0}^{t_f} \sum_i(\tau_i(s)\nu_i)^2\mathrm{d}t = w_e\int_{s_0}^{s_f} \sum_i(\tau_i(s)\nu_i)^2\frac{1}{\sqrt{a(s)}}\mathrm{d}s$ where `nu_i = normalize[i]`.  
33    /// + Discrete (COPP2): `J_th = 2*w_e*sum_{k=0}^{n-2} (s[k+1]-s[k])/(sqrt(a[k])+sqrt(a[k+1])) * sum_i (tau[i][k]*normalize[i])^2`.  
34    /// + Discrete (COPP3): `J_th = w_e*sum_k weight_a_torque[k] * sum_i (tau[i][k]*normalize[i])^2`.
35    ThermalEnergy(f64, &'a [f64]),
36    /// Total-variation of torque objective.  
37    /// + Continuous: $J_{\mathrm{tv}} = w_v\sum_i \int_{0}^{t_f} \left|\frac{d\tau_i}{\mathrm{d}s}(s)\right|\nu_i\mathrm{d}t = w_v\sum_i \int_{s_0}^{s_f} \left|\frac{d\tau_i}{\mathrm{d}s}(s)\right|\nu_i\mathrm{d}s$.
38    /// + Discrete:  `J_tv = w_v*sum_i sum_k |tau[i][k+1]-tau[i][k]|*normalize[i]`
39    TotalVariationTorque(f64, &'a [f64]),
40    /// Linear objective over `a` and `b`.  
41    /// + Continuous: $J_{\mathrm{lin}} = w_l\int_{s_0}^{s_f}(\alpha(s)a(s)+\beta(s)b(s))\mathrm{d}s$.
42    /// + Discrete (COPP2):
43    ///   - `alpha.len()==n`, `beta.len()==n-1`
44    ///   - `J_lin = w_l*( sum_{k=0}^{n-1} alpha[k]*a[k] + sum_{k=0}^{n-2} beta[k]*b[k] )`
45    /// + Discrete (COPP3):
46    ///   - `alpha.len()==beta.len()==n`
47    ///   - `J_lin = w_l*sum_{k=0}^{n-1} (alpha[k]*a[k] + beta[k]*b[k])`
48    Linear(f64, &'a [f64], &'a [f64]),
49}

51/// Validate objective terms for COPP2.  
52/// `s_len` must be the number of grid points in the closed interval `[idx_s_start, idx_s_final]`, i.e. `idx_s_final - idx_s_start + 1`.  
53/// Rules:  
54/// - `objectives` must be non-empty.  
55/// - all weights must be non-negative and finite.  
56/// - for `ThermalEnergy` / `TotalVariationTorque`, `normalize.len() == dim`.  
57/// - all normalize entries must be non-negative and finite.  
58/// - for `Linear`, `alpha.len() == s_len` and `beta.len() + 1 == s_len`.
59#[inline]
60pub(crate) fn validate_copp2_objectives(
61    function_name: &str,
62    objectives: &[CoppObjective<'_>],
63    dim: usize,
64    s_len: usize,
65) -> Result<(), CoppError> {
66    if objectives.is_empty() {
67        return Err(CoppError::InvalidInput(
68            function_name.into(),
69            "objectives must not be empty".into(),
70        ));
71    }
72
73    for (i, objective) in objectives.iter().enumerate() {
74        match objective {
75            CoppObjective::Time(weight) => {
76                if weight.is_nan() || weight.is_infinite() || *weight < 0.0 {
77                    return Err(CoppError::InvalidInput(
78                        function_name.into(),
79                        format!(
80                            "objectives[{i}]::Time(weight={weight}) must be finite and non-negative"
81                        ),
82                    ));
83                }
84            }
85            CoppObjective::ThermalEnergy(weight, normalize)
86            | CoppObjective::TotalVariationTorque(weight, normalize) => {
87                if weight.is_nan() || weight.is_infinite() || *weight < 0.0 {
88                    return Err(CoppError::InvalidInput(
89                        function_name.into(),
90                        format!("objectives[{i}] weight={weight} must be finite and non-negative"),
91                    ));
92                }
93                if normalize.len() != dim {
94                    return Err(CoppError::InvalidInput(
95                        function_name.into(),
96                        format!(
97                            "objectives[{i}] normalize.len()={} must equal robot dim={dim}",
98                            normalize.len()
99                        ),
100                    ));
101                }
102                for (j, v) in normalize.iter().enumerate() {
103                    if v.is_nan() || v.is_infinite() || *v < 0.0 {
104                        return Err(CoppError::InvalidInput(
105                            function_name.into(),
106                            format!(
107                                "objectives[{i}] normalize[{j}]={v} must be finite and non-negative"
108                            ),
109                        ));
110                    }
111                }
112            }
113            CoppObjective::Linear(weight, alpha, beta) => {
114                if weight.is_nan() || weight.is_infinite() || *weight < 0.0 {
115                    return Err(CoppError::InvalidInput(
116                        function_name.into(),
117                        format!(
118                            "objectives[{i}]::Linear(weight={weight}) must be finite and non-negative"
119                        ),
120                    ));
121                }
122                if alpha.len() != s_len || beta.len() + 1 != s_len {
123                    return Err(CoppError::InvalidInput(
124                        function_name.into(),
125                        format!(
126                            "objectives[{i}]::Linear length mismatch: alpha.len()={}, beta.len()={}, expected alpha.len()={s_len}, beta.len()={}",
127                            alpha.len(),
128                            beta.len(),
129                            s_len - 1
130                        ),
131                    ));
132                }
133            }
134        }
135    }
136
137    Ok(())
138}

140/// Validate objective terms for COPP3.  
141/// `s_len` must be the number of grid points used by TOPP3/COPP3 (typically `a_linear.len()`).  
142/// Rules:  
143/// - `objectives` must be non-empty.  
144/// - all weights must be non-negative and finite.  
145/// - for `ThermalEnergy` / `TotalVariationTorque`, `normalize.len() == dim`.  
146/// - all normalize entries must be non-negative and finite.  
147/// - for `Linear`, `alpha.len() == s_len` and `beta.len() == s_len`.
148#[inline]
149pub(crate) fn validate_copp3_objectives(
150    function_name: &str,
151    objectives: &[CoppObjective<'_>],
152    dim: usize,
153    s_len: usize,
154) -> Result<(), CoppError> {
155    if objectives.is_empty() {
156        return Err(CoppError::InvalidInput(
157            function_name.into(),
158            "objectives must not be empty".into(),
159        ));
160    }
161
162    for (i, objective) in objectives.iter().enumerate() {
163        match objective {
164            CoppObjective::Time(weight) => {
165                if weight.is_nan() || weight.is_infinite() || *weight < 0.0 {
166                    return Err(CoppError::InvalidInput(
167                        function_name.into(),
168                        format!(
169                            "objectives[{i}]::Time(weight={weight}) must be finite and non-negative"
170                        ),
171                    ));
172                }
173            }
174            CoppObjective::ThermalEnergy(weight, normalize)
175            | CoppObjective::TotalVariationTorque(weight, normalize) => {
176                if weight.is_nan() || weight.is_infinite() || *weight < 0.0 {
177                    return Err(CoppError::InvalidInput(
178                        function_name.into(),
179                        format!("objectives[{i}] weight={weight} must be finite and non-negative"),
180                    ));
181                }
182                if normalize.len() != dim {
183                    return Err(CoppError::InvalidInput(
184                        function_name.into(),
185                        format!(
186                            "objectives[{i}] normalize.len()={} must equal robot dim={dim}",
187                            normalize.len()
188                        ),
189                    ));
190                }
191                for (j, v) in normalize.iter().enumerate() {
192                    if v.is_nan() || v.is_infinite() || *v < 0.0 {
193                        return Err(CoppError::InvalidInput(

195                            format!(
196                                "objectives[{i}] normalize[{j}]={v} must be finite and non-negative"
197                            ),
198                        ));
199                    }
200                }
201            }
202            CoppObjective::Linear(weight, alpha, beta) => {
203                if weight.is_nan() || weight.is_infinite() || *weight < 0.0 {
204                    return Err(CoppError::InvalidInput(
205                        function_name.into(),
206                        format!(
207                            "objectives[{i}]::Linear(weight={weight}) must be finite and non-negative"
208                        ),
209                    ));
210                }
211                if alpha.len() != s_len || beta.len() != s_len {
212                    return Err(CoppError::InvalidInput(
213                        function_name.into(),
214                        format!(
215                            "objectives[{i}]::Linear length mismatch: alpha.len()={}, beta.len()={}, expected alpha.len()={s_len}, beta.len()={s_len}",
216                            alpha.len(),
217                            beta.len(),
218                        ),
219                    ));
220                }
221            }
222        }
223    }
224
225    Ok(())
