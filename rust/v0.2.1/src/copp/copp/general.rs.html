1//! General shared definitions and numeric utilities for TOPP/COPP flows.
2//!
3//! # Method identity
4//! This module hosts lightweight, cross-cutting primitives that are reused by
5//! multiple solver/formulation layers, including:
6//! - interpolation policy descriptors for output trajectory sampling;
7//! - tolerant floating-point comparison helpers used by feasibility and
8//!   convergence checks.
9//!
10//! # Design notes
11//! - Types here are intentionally small and dependency-free.
12//! - Approximate comparison uses a mixed tolerance
13//!   $\max(\text{abs\_tol},\ \text{rel\_tol}\cdot\max(|x_1|,|x_2|))$.
14//! - `approx_order()` is preferred over direct equality checks when decisions
15//!   depend on floating-point values near boundaries.

17/// Time-grid policy used when interpolating path-parameterization outputs.
18///
19/// This enum describes how target sample times are provided to interpolation
20/// routines after a trajectory has been parameterized.
21///
22/// # Output behavior
23/// Interpolation helpers such as
24/// [`t_to_s_topp2`](crate::solver::topp2_ra::t_to_s_topp2) and
25/// [`t_to_s_topp3`](crate::solver::topp3_socp::t_to_s_topp3) preserve the
26/// requested sampling policy:
27/// - [`UniformTimeGrid`](InterpolationMode::UniformTimeGrid) emits samples
28///   `t0 + i * dt` up to the final time, and may append the exact final sample
29///   when `include_final` is `true`.
30/// - [`NonUniformTimeGrid`](InterpolationMode::NonUniformTimeGrid) returns one
31///   output value per caller-provided time stamp.
32/// - Samples outside the computed trajectory time range are reported as `NaN`
33///   by the interpolation helpers.
34pub enum InterpolationMode<'a> {
35    /// Uniform sampling grid.
36    ///
37    /// # Tuple fields
38    /// - `t0`: time stamp of the first path station (`s[0]`).
39    /// - `dt`: constant sampling period (`dt > 0` expected by callers).
40    /// - `include_final`: handling of non-integer final step.
41    ///
42    /// # Final-sample policy
43    /// If final time `t_final` is not an integer multiple of `dt` from `t0`:
44    /// - `include_final = true`: append one final sample exactly at `t_final`.
45    /// - `include_final = false`: stop at `t0 + n\,dt`, where `n` is the
46    ///   largest integer satisfying `t0 + n\,dt \le t_final`.
47    UniformTimeGrid(f64, f64, bool),
48
49    /// User-provided non-uniform sampling grid.
50    ///
51    /// # Tuple fields
52    /// - `t_samples`: strictly increasing time stamps.
53    ///
54    /// # Contract
55    /// Callers should provide an increasing grid and ensure the first sample is
56    /// not earlier than the interpolation start time.
57    NonUniformTimeGrid(&'a [f64]),
58}

60/// Approximate ordering relation for two floating-point values.
61///
62/// Returned by [`approx_order()`] when comparing `x1` and `x2` under mixed
63/// absolute/relative tolerance.
64pub(crate) enum ApproxOrdering {
65    /// `x1 < x2` under tolerance-aware comparison.
66    Less,
67    /// `x1 \approx x2` within tolerance band.
68    Equal,
69    /// `x1 > x2` under tolerance-aware comparison.
70    Greater,
71}
72
73/// Compute the mixed absolute/relative comparison threshold.
74///
75/// # Formula
76/// `threshold = max(abs_tol, rel_tol * max(|x1|, |x2|))`
77///
78/// # Parameters
79/// - `x1`, `x2`: values to be compared.
80/// - `abs_tol`: absolute tolerance component.
81/// - `rel_tol`: relative tolerance component.
82///
83/// # Returns
84/// A non-negative scalar used as symmetric comparison band around zero for
85/// `dx = x1 - x2`.
86///
87/// # Naming note
88/// Function name keeps historical spelling (`threhold_approx`) for API
89/// compatibility.
90#[inline(always)]
91pub(crate) fn threhold_approx(x1: f64, x2: f64, abs_tol: f64, rel_tol: f64) -> f64 {
92    abs_tol.max(rel_tol * x1.abs().max(x2.abs()))
93}
94
95/// Compare two floating-point values with mixed tolerance and return ordering.
96///
97/// # Decision rule
98/// Let `dx = x1 - x2` and
99/// `threshold = threhold_approx(x1, x2, abs_tol, rel_tol)`.
100///
101/// - return [`ApproxOrdering::Greater`] if `dx > threshold`;
102/// - return [`ApproxOrdering::Less`] if `dx < -threshold`;
103/// - otherwise return [`ApproxOrdering::Equal`].
104///
105/// # Parameters
106/// - `x1`, `x2`: values to compare.
107/// - `abs_tol`: absolute tolerance.
108/// - `rel_tol`: relative tolerance.
109///
110/// # Returns
111/// Tolerance-aware ordering relation between `x1` and `x2`.
112#[inline(always)]
113pub(crate) fn approx_order(x1: f64, x2: f64, abs_tol: f64, rel_tol: f64) -> ApproxOrdering {
114    let threshold = threhold_approx(x1, x2, abs_tol, rel_tol);
115    let dx = x1 - x2;
116    if dx > threshold {
117        ApproxOrdering::Greater
118    } else if dx < -threshold {
119        ApproxOrdering::Less
120    } else {
121        ApproxOrdering::Equal
122    }
123}
